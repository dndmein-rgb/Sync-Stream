import express from "express";
import { getReadyStatus } from "./infrastructure/ready.js";
import { corsMiddleware } from "./middlewares/cors.middleware.js";
import { requestLogger } from "./middlewares/req.middleware.js";
import { errorMiddleware } from "./middlewares/error.middleware.js";
import authRoutes from "./modules/auth/auth.route.js";
import roomRoutes from "./modules/room/room.route.js";

export function createApp() {
  const app = express();

  app.use(corsMiddleware);
  app.use(express.json());
  app.use(requestLogger);

  app.get("/health", (_req, res) => {
    res.status(200).json({
      status: "ok",
      service: "syncstream-api",
      timestamp: new Date().toISOString(),
    });
  });

  app.get("/ready", async (_req, res) => {
    const result = await getReadyStatus();
    res.status(result.ready ? 200 : 503).json({
      status: result.ready ? "ready" : "not_ready",
      service: "syncstream-api",
      checks: result.checks,
      timestamp: new Date().toISOString(),
    });
  });

  app.use("/auth", authRoutes);
  app.use("/rooms", roomRoutes);

  app.use(errorMiddleware);
  return app;
}

import { createServer } from "node:http";
import { createApp } from "./app.js";
import { config } from "./config/index.js";
import { logger } from "./config/logger.js";
import { prisma } from "./infrastructure/prisma.js";
import { createSocketServer } from "./realtime/socket-server.js";

async function main() {
  const app = createApp();
  const httpServer = createServer(app);

  // Attach Socket.IO to the same HTTP server
  const io = createSocketServer(httpServer);

  httpServer.listen(config.PORT, () => {
    logger.info(`syncstream-api listening on port ${config.PORT}`);
    logger.info(`Socket.IO path: ${config.SOCKET_PATH}`);
  });

  const shutdown = async (signal: string) => {
    logger.info(`Shutting down API (${signal})`);

    // Stop accepting new connections
    io.close();

    httpServer.close(async () => {
      try {
        await prisma.$disconnect();
        logger.info("API shutdown complete");
        process.exit(0);
      } catch (err) {
        logger.error(err, "Error during API shutdown");
        process.exit(1);
      }
    });

    // Force exit if drain takes too long
    setTimeout(() => process.exit(1), 15_000).unref();
  };

  process.on("SIGINT", () => void shutdown("SIGINT"));
  process.on("SIGTERM", () => void shutdown("SIGTERM"));
}

main().catch((err) => {
  logger.error(err, "Failed to start server");
  process.exit(1);
});

import pino from "pino";
import { config } from "./index.js";

const baseLogger = pino({
  level: config.NODE_ENV === "development" ? "debug" : config.LOG_LEVEL,
  ...(config.NODE_ENV === "development" && {
    transport: {
      target: "pino-pretty",
      options: {
        colorize: true,
        translateTime: "SYS:standard",
        ignore: "pid,hostname",
      },
    },
  }),
  base: {
    service: config.SERVICE_NAME,
  },
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) return {};
  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }
  if (typeof meta === "object") return meta as Record<string, unknown>;
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    if (typeof msgOrMeta === "string") {
      baseLogger[level](toMergingObject(maybeMetaOrMsg), msgOrMeta);
      return;
    }
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;
    if (msg) baseLogger[level](obj, msg);
    else baseLogger[level](obj);
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

export type ReadyCheck = {

  ok: boolean;
  latencyMs?: number;
  error?: string;
};

export type ReadyResult = {
  ready: boolean;
  checks: ReadyCheck[];
};

async function checkPostgres(): Promise<ReadyCheck> {
  const start = Date.now();
  const pool = new pg.Pool({
    connectionString: config.DATABASE_URL,
    max: 1,
    connectionTimeoutMillis: 2_000,
  });
  try {
    await pool.query("SELECT 1");
    return { name: "postgres", ok: true, latencyMs: Date.now() - start };
  } catch (err) {
    return {
      name: "postgres",
      ok: false,
      latencyMs: Date.now() - start,
      error: err instanceof Error ? err.message : String(err),
    };
  } finally {
    await pool.end().catch(() => undefined);
  }
}

    ready: pgCheck.ok,

  };

  NextFunction,

export const authenticate = async(
  req: Request,
  _res: Response,
  next: NextFunction,
): Promise<void> => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader) {
      return next(
        new UnauthorizedError("Authentication required"),
      );
    }

    if (!authHeader.startsWith("Bearer ")) {
      return next(
        new UnauthorizedError(
          "Invalid authentication header format",
        ),
      );
    }

    const accessToken = authHeader.split(" ")[1];

    if (!accessToken) {
      return next(
        new UnauthorizedError("Access token missing"),
      );
    }
    const payload = verifyAccessToken(accessToken);
  const user=await authService.resolveUser(payload.sub)

    req.user = user

    return next();
  } catch (error) {
    if (error instanceof jwt.TokenExpiredError) {
      return next(
        new UnauthorizedError("Access token expired"),
      );
    }

    if (error instanceof jwt.JsonWebTokenError) {
      return next(
        new UnauthorizedError("Invalid access token"),
      );
    }

    return next(error);
  }
};

import type { NextFunction, Request, Response } from "express";
import { ZodType } from "zod";
import type { ParamsDictionary } from "express-serve-static-core";
import { BadRequestError } from "@syncstream/shared-utils";

type ValidationTarget = "body" | "params" | "query";

export const validate =
  <T>(schema: ZodType<T>, target: ValidationTarget = "body") =>
  (req: Request, res: Response, next: NextFunction): void => {
    const result = schema.safeParse(req[target]);
    if (!result.success) {
      const message = result.error.issues
        .map(({ path, message }) => `${path.join(".")}:${message}`)
        .join(",");
      return next(new BadRequestError(message));
    }
    switch (target) {
      case "body":
        req.body = result.data;
        break;

      case "params":
        req.params = result.data as ParamsDictionary;
        break;

      case "query":
        res.locals.validatedQuery = result.data;
        break;
    }

    next();
  };

import jwt, { SignOptions } from "jsonwebtoken";
import { config } from "../../config/index.js";
import crypto from "node:crypto";
import { UnauthorizedError } from "@syncstream/shared-utils";

export type AccessPayload = {
  sub: string;
  typ: "access";
};

export function signAccessToken(userId: string): string {
  return jwt.sign(
    { sub: userId, typ: "access" } satisfies AccessPayload,
    config.JWT_ACCESS_SECRET,
    { expiresIn: config.JWT_ACCESS_EXPIRES_IN as SignOptions["expiresIn"] },
  );
}


export function verifyAccessToken(token: string): AccessPayload {
  try {
    const payload = jwt.verify(token, config.JWT_ACCESS_SECRET) as AccessPayload;
    if (payload.typ !== "access" || !payload.sub) {
      throw new UnauthorizedError("Invalid access token");
    }
    return payload;
  } catch {
    throw new UnauthorizedError("Invalid or expired access token");
  }
}

/** Opaque refresh token (not a JWT) â€” store only the hash. */
export function generateRefreshToken(): string {
  return crypto.randomBytes(48).toString("base64url");
}

export function hashToken(token: string): string {
  return crypto.createHash("sha256").update(token).digest("hex");
}

export function refreshExpiresAt(): Date {
  const match = /^(\d+)([dhms])$/.exec(config.JWT_REFRESH_EXPIRES_IN);
  const now = Date.now();
  if (!match) {
    return new Date(now + 7 * 24 * 60 * 60 * 1000);
  }
  const n = Number(match[1]);
  const unit = match[2];
  const ms =
    unit === "d" ? n * 86400000 :
    unit === "h" ? n * 3600000 :
    unit === "m" ? n * 60000 :
    n * 1000;
  return new Date(now + ms);
}

import {
  ConflictError,
  UnauthorizedError,
  ValidationError,
} from "@syncstream/shared-utils";
import { AuthInterface } from "./auth.interface.js";
import { LoginBody, RegisterBody } from "./auth.schema.js";
import bcrypt from "bcryptjs";
import { config } from "../../config/index.js";
import {
  generateRefreshToken,
  hashToken,
  refreshExpiresAt,
  signAccessToken,
} from "./auth.helper.js";
import { SafeUser, TokenPair } from "./auth.types.js";

export class AuthService {
  constructor(private readonly repo: AuthInterface) {}

  async register(input: RegisterBody) {
    const { email, name, password } = input;
    const existing = await this.repo.findByEmail(email);
    if (existing) {
      throw new ConflictError("Email already registered");
    }
    const passwordHash = await bcrypt.hash(password, config.BCRYPT_ROUNDS);
    const user = await this.repo.createUser({
      name,
      email,
      passwordHash,
    });
    const tokens = await this.issueTokens(user.id);
    return { user, tokens };
  }
  async login(input: LoginBody) {
    const user = await this.repo.findByEmail(input.email);
    if (!user) {
      throw new UnauthorizedError("Invalid email or password");
    }
    const isPasswordValid = await bcrypt.compare(
      input.password,
      user.passwordHash,
    );

    if (!isPasswordValid) {
      throw new UnauthorizedError("Invalid email or password");
    }

    const tokens = await this.issueTokens(user.id);
    return {
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        avatarUrl: user.avatarUrl,
      },
      tokens,
    };
  }

  async refresh(refreshToken: string): Promise<TokenPair> {
    if (!refreshToken) {
      throw new ValidationError("refreshToken is required");
    }

    const tokenHash = hashToken(refreshToken);
    const token = await this.repo.findValidRefreshToken(tokenHash);
    if (!token) {
      throw new UnauthorizedError("Invalid or expired refresh token");
    }

    // Rotate: revoke old, issue new pair
    await this.repo.revokeRefreshToken(token.id);
    return this.issueTokens(token.userId);
  }

  async logout(refreshToken: string | undefined): Promise<void> {
    if (!refreshToken) return;
    const tokenHash = hashToken(refreshToken);
    const row = await this.repo.findValidRefreshToken(tokenHash);
    if (row) {
      await this.repo.revokeRefreshToken(row.id);
    }
  }

  async logoutAll(userId: string): Promise<void> {
    await this.repo.revokeAllForUser(userId);
  }

  async getMe(userId: string): Promise<SafeUser> {
    const user = await this.repo.findById(userId);
    if (!user) {
      throw new UnauthorizedError("User not found");
    }
    return user;
  }

  /** Used by HTTP + Socket middleware after JWT verify. */
  async resolveUser(userId: string): Promise<SafeUser> {
    return this.getMe(userId);
  }
  private async issueTokens(userId: string): Promise<TokenPair> {
    const accessToken = signAccessToken(userId);
    const refreshToken = generateRefreshToken();
    const tokenHash = hashToken(refreshToken);

    await this.repo.createRefreshToken({
      userId,
      tokenHash,
      expiresAt: refreshExpiresAt(),
    });

    return {
      accessToken,
      refreshToken,
      expiresIn: config.JWT_ACCESS_EXPIRES_IN,
    };
  }
}

import {
  PlaybackCommand,
  PlaybackState,
  RoomRole,
} from "@syncstream/contracts";
import { RoomPlayback } from "./playback.types.js";
import { ForbiddenError, ValidationError } from "@syncstream/shared-utils";
import { assertCan } from "../room/room.permissions.js";

const MAX_RECENT_COMMANDS = 64; // per room â€“ enough to stop network retries

/**
 * Single-process authoritative playback state.
 * Key = roomId.
 *
 * Later (Phase 6) this Map moves behind Redis + Lua, but the public API
 * of the service stays the same.
 */

export class PlaybackService {
  private readonly states = new Map<string, RoomPlayback>();
  /** roomId â†’ recent commandIds (for deduplication) */
  private readonly recentCommands = new Map<string, Set<string>>();

  /** Returns current state or null if the room has never selected a video. */
  getState(roomId: string): PlaybackState | null {
    const s = this.states.get(roomId);
    return s ? { ...s } : null;
  }

  /**
   * Apply a host/co-host command.
   * Throws AppError subclasses on validation / permission failure.
   */
  applyCommand(
    roomId: string,
    userId: string,
    role: RoomRole,
    cmd: PlaybackCommand,
  ) {
    // 1. Permission
    assertCan(role, "PLAYBACK_CONTROL");
    // 2. Deduplication
    if (this.isDuplicate(roomId, cmd.commandId)) {
      const current = this.states.get(roomId);
      if (!current) {
        // very unlikely â€“ command was accepted earlier but state gone
        throw new ValidationError("Duplicate command but no state exists");
      }
      return { state: { ...current }, duplicate: true };
    }
    const now = Date.now();
    let current = this.states.get(roomId);

    // 3. State transition
    let next: RoomPlayback;
    switch (cmd.action) {
      case "CHANGE_VIDEO": {
        if (!cmd.videoId) {
          throw new ValidationError("videoId is required for CHANGE_VIDEO");
        }

        next = {
          videoId: cmd.videoId,
          status: "PAUSED",
          position: 0,
          updatedAt: now,
          playbackRate: 1,
          sequence: current ? current.sequence + 1 : 1,
        };
        break;
      }
      case "PLAY": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        // If already playing we still accept the command (idempotent) but
        // re-anchor the position so late clients stay in sync.
        const position =
          cmd.position !== undefined
            ? cmd.position
            : this.expectedPosition(current, now);
        next = {
          ...current,
          status: "PLAYING",
          position,
          updatedAt: now,
          sequence: current.sequence + 1,
        };
        break;
      }
      case "PAUSE": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        const position = this.expectedPosition(current, now);
        next = {
          ...current,
          status: "PAUSED",
          position,
          updatedAt: now,
          sequence: current.sequence + 1,
        };
        break;
      }
      case "SEEK": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        if (cmd.position === undefined) {
          throw new ValidationError("position is required for SEEK");
        }
        if (cmd.position < 0) {
          throw new ValidationError("position cannot be negative");
        }
        next = {
          ...current,
          position: cmd.position,
          updatedAt: now,
          sequence: current.sequence + 1,
          // keep current status (playing or paused)
        };
        break;
      }
      default: {
        // exhaustive check
        const _exhaustive: never = cmd.action;
        throw new ValidationError(`Unknown action: ${_exhaustive}`);
      }
    }
    // 4. Persist + remember commandId
    this.states.set(roomId, next);
    this.rememberCommand(roomId, cmd.commandId);

    return { state: { ...next }, duplicate: false };
  }
  /** Remove state when a room is deleted (called from room service later). */
  clearRoom(roomId: string): void {
    this.states.delete(roomId);
    this.recentCommands.delete(roomId);
  }

  // â”€â”€ helpers â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

  /**
   * Expected playback position at time `now` according to the formula:
   *   PLAYING â†’ position + rate * (now - updatedAt) / 1000
   *   PAUSED  â†’ position
   */
  expectedPosition(state: PlaybackState, now: number): number {
    if (state.status === "PAUSED") return state.position;
    const elapsedSec = (now - state.updatedAt) / 1000;
    return Math.max(0, state.position + state.playbackRate * elapsedSec);
  }
  private isDuplicate(roomId: string, commandId: string): boolean {
    const set = this.recentCommands.get(roomId);
    return set ? set.has(commandId) : false;
  }
  private rememberCommand(roomId: string, commandId: string): void {
    let set = this.recentCommands.get(roomId);
    if (!set) {
      set = new Set();
      this.recentCommands.set(roomId, set);
    }
    set.add(commandId);
    // simple bound â€“ drop oldest when over limit (Set iteration order is insertion order)
    if (set.size > MAX_RECENT_COMMANDS) {
      const first = set.values().next().value;
      if (first !== undefined) set.delete(first);
    }
  }
}

const ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ";

export function generateRoomCode(length = 8):string {
  const bytes = crypto.randomBytes(length)
  let code = "";
    for (let i = 0; i < length; i++) {
      code += ALPHABET[bytes[i]! % ALPHABET.length];
    }
    return code;

import {
  ConflictError,
  ForbiddenError,
  NotFoundError,
} from "@syncstream/shared-utils";
import { prisma } from "../../infrastructure/prisma.js";
import type { RoomInterface } from "./room.interface.js";
import type {
  CreateRoomInput,
  MemberRecord,
  RoomRecord,
  UpdateRoomInput,
} from "./room.types.js";
import type { RoomRole } from "@syncstream/contracts";
import { generateRoomCode } from "./room.helper.js";

function toRoomRecord(row: {
  id: string;
  code: string;
  name: string;
  visibility: string;
  hostId: string;
  maxMembers: number;
  createdAt: Date;
  updatedAt: Date;
}): RoomRecord {
  return {
    id: row.id,
    code: row.code,
    name: row.name,
    visibility: row.visibility as RoomRecord["visibility"],
    hostId: row.hostId,
    maxMembers: row.maxMembers,
    createdAt: row.createdAt,
    updatedAt: row.updatedAt,
  };
}

export class RoomRepository implements RoomInterface {
  async createRoomWithHost(data: CreateRoomInput): Promise<RoomRecord> {
    const maxAttempts = 5;
    for (let attempt = 0; attempt < maxAttempts; attempt++) {
      const code = generateRoomCode(8);
      try {
        const room = await prisma.$transaction(async (tx) => {
          const created = await tx.room.create({
            data: {
              code,
              name: data.name,
              visibility: data.visibility ?? "PRIVATE",
              hostId: data.hostId,
              maxMembers: data.maxMembers ?? 10,
              members: {
                create: {
                  userId: data.hostId,
                  role: "HOST",
                },
              },
            },
          });
          return created;
        });
        return toRoomRecord(room);
      } catch (err: unknown) {
        // Unique violation on code â€” retry
        const codeConflict =
          typeof err === "object" &&
          err !== null &&
          "code" in err &&
          (err as { code?: string }).code === "P2002";
        if (codeConflict && attempt < maxAttempts - 1) continue;
        throw err;
      }
    }
    throw new ConflictError("Could not allocate a unique room code");
  }

  async findById(id: string): Promise<RoomRecord | null> {
    const row = await prisma.room.findUnique({ where: { id } });
    return row ? toRoomRecord(row) : null;
  }

  async findByCode(code: string): Promise<RoomRecord | null> {
    const row = await prisma.room.findUnique({
      where: { code: code.toUpperCase() },
    });
    return row ? toRoomRecord(row) : null;
  }

  async updateRoom(id: string, data: UpdateRoomInput): Promise<RoomRecord> {
    const row = await prisma.room.update({
      where: { id },
      data: {
        ...(data.name !== undefined ? { name: data.name } : {}),
        ...(data.visibility !== undefined ? { visibility: data.visibility } : {}),
        ...(data.maxMembers !== undefined ? { maxMembers: data.maxMembers } : {}),
      },
    });
    return toRoomRecord(row);
  }

  async deleteRoom(id: string): Promise<void> {
    await prisma.room.delete({ where: { id } });
  }

  async countMembers(roomId: string): Promise<number> {
    return prisma.roomMember.count({ where: { roomId } });
  }

  async findMembership(roomId: string, userId: string) {
    return prisma.roomMember.findUnique({
      where: { roomId_userId: { roomId, userId } },
      select: { roomId: true, userId: true, role: true, joinedAt: true },
    });
  }

  async listMembers(roomId: string): Promise<MemberRecord[]> {
    const rows = await prisma.roomMember.findMany({
      where: { roomId },
      orderBy: { joinedAt: "asc" },
      include: {
        user: {
          select: { id: true, name: true, avatarUrl: true },
        },
      },
    });
    return rows.map((r) => ({
      roomId: r.roomId,
      userId: r.userId,
      role: r.role as RoomRole,
      joinedAt: r.joinedAt,
      user: r.user,
    }));
  }

  async joinAsViewer(
    roomId: string,
    userId: string,
  ): Promise<{ membership: { role: RoomRole; joinedAt: Date }; created: boolean }> {
    return prisma.$transaction(async (tx) => {
      const existing = await tx.roomMember.findUnique({
        where: { roomId_userId: { roomId, userId } },
      });
      if (existing) {
        return {
          membership: {
            role: existing.role as RoomRole,
            joinedAt: existing.joinedAt,
          },
          created: false,
        };
      }

      const room = await tx.room.findUnique({ where: { id: roomId } });
      if (!room) {
        throw new NotFoundError("Room", roomId);
      }

      const count = await tx.roomMember.count({ where: { roomId } });
      if (count >= room.maxMembers) {
        throw new ConflictError("Room is full");
      }

      const created = await tx.roomMember.create({
        data: {
          roomId,
          userId,
          role: "VIEWER",
        },
      });

      return {
        membership: {
          role: created.role as RoomRole,
          joinedAt: created.joinedAt,
        },
        created: true,
      };
    });
  }

  async removeMember(roomId: string, userId: string): Promise<void> {
    await prisma.roomMember.delete({
      where: { roomId_userId: { roomId, userId } },
    });
  }

  async setMemberRole(roomId: string, userId: string, role: RoomRole): Promise<void> {
    await prisma.roomMember.update({
      where: { roomId_userId: { roomId, userId } },
      data:{role}
    })
  }
  async transferHost(roomId: string, fromUserId: string, toUserId: string): Promise<void> {
    await prisma.$transaction(async (tx) => {
      const room = await tx.room.findUnique({ where: { id: roomId } });
          if (!room) throw new NotFoundError("Room", roomId);
          if (room.hostId !== fromUserId) {
            throw new ForbiddenError("Only the current host can transfer");
          }
      
          const target = await tx.roomMember.findUnique({
            where: { roomId_userId: { roomId, userId: toUserId } },
          });
          if (!target) {
            throw new NotFoundError("Member", toUserId);
      }
      await tx.roomMember.update({
            where: { roomId_userId: { roomId, userId: fromUserId } },
            data: { role: "VIEWER" },
          });
          await tx.roomMember.update({
            where: { roomId_userId: { roomId, userId: toUserId } },
            data: { role: "HOST" },
          });
          await tx.room.update({
            where: { id: roomId },
            data: { hostId: toUserId },
          });
    })
  }
}

export type PresencePort = {
  isOnline(roomId: string, userId: string): boolean;
  getOnlineUserIds(roomId: string): Set<string>;
};

export class RoomService {
  constructor(
    private readonly repo: RoomInterface,
    private readonly presence: PresencePort,
  ) { }

  private toRoomView(
    room: {
      id: string;
      code: string;
      name: string;
      visibility: RoomView["visibility"];
      hostId: string;
      maxMembers: number;
      createdAt: Date;
      updatedAt: Date;
    },

    return {

      name: room.name,
      visibility: room.visibility,
      hostId: room.hostId,
      maxMembers: room.maxMembers,
      memberCount,
      createdAt: room.createdAt.toISOString(),
      updatedAt: room.updatedAt.toISOString(),
      ...(myRole ? { myRole } : {}),
    };
  }

  private toMemberViews(
    roomId: string,
    members: Awaited<ReturnType<RoomInterface["listMembers"]>>,
  ): MemberView[] {
    return members.map((m) => ({
      userId: m.userId,
      name: m.user.name,
      avatarUrl: m.user.avatarUrl,
      role: m.role,
      online: this.presence.isOnline(roomId, m.userId),
      joinedAt: m.joinedAt.toISOString(),
    }));
  }

  async create(hostId: string, body: CreateRoomBody) {
    const room = await this.repo.createRoomWithHost({
      hostId,
      name: body.name,
      maxMembers: body.maxMembers,
      visibility: body.visibility,
    });
    return this.toRoomView(room, 1, "HOST");
  }
  async getById(roomId: string, requesterId: string) {
    const room = await this.repo.findById(roomId);
    if (!room) {
      if (!room) throw new NotFoundError("Room", roomId);
    }
    const membership = await this.repo.findMembership(roomId, requesterId);
    if (!membership) {
      throw new ForbiddenError("You are not a member of this room");
    }
    const count = await this.repo.countMembers(roomId);
    return this.toRoomView(room, count, membership.role);
  }
  async getMembers(roomId: string, requesterId: string): Promise<MemberView[]> {
    const membership = await this.repo.findMembership(roomId, requesterId);
    if (!membership) {
      throw new ForbiddenError("You are not a member of this room");
    }
    const members = await this.repo.listMembers(roomId);
    return this.toMemberViews(roomId, members);
  }

   * Private rooms require correct code unless already a member.
   */

    userId: string,
    opts: { roomId?: string; code?: string },
  ): Promise<{ room: RoomView; members: MemberView[]; created: boolean }> {
    let room =
      opts.roomId != null
        ? await this.repo.findById(opts.roomId)
        : opts.code != null
          ? await this.repo.findByCode(opts.code)
          : null;
    if (!room && opts.code && opts.roomId) {
      // join by id + verify code
      room = await this.repo.findById(opts.roomId);
    }
    if (!room) throw new NotFoundError("Room");
    const existing = await this.repo.findMembership(room.id, userId);
    
    if (!existing) {
      if (room.visibility === "PRIVATE") {
        const code = opts.code?.toUpperCase();
        if (!code || code !== room.code) {
          throw new ForbiddenError("Valid room code required");
        }
      }
      const { membership, created } = await this.repo.joinAsViewer(room.id, userId);
      const count = await this.repo.countMembers(room.id);
      const members = this.toMemberViews(room.id, await this.repo.listMembers(room.id));
      return {
        room: this.toRoomView(room, count, membership.role),
        members,
        created,
      };
    }
    // Already member â€” idempotent (reconnect-friendly)
        if (
          room.visibility === "PRIVATE" &&
          opts.code &&
          opts.code.toUpperCase() !== room.code
        ) {
          // optional: still allow if already member even with wrong code
        }
        const count = await this.repo.countMembers(room.id);
            const members = this.toMemberViews(room.id, await this.repo.listMembers(room.id));
            return {
              room: this.toRoomView(room, count, existing.role),
              members,
              created: false,
            };
  }
  async joinByCode(userId: string, body: JoinByCodeBody) {
      return this.join(userId, { code: body.code });
    }
  
    async joinById(userId: string, roomId: string, body: JoinByIdBody) {

    }

    const room = await this.repo.findById(roomId);
        if (!room) throw new NotFoundError("Room", roomId);
    
        const membership = await this.repo.findMembership(roomId, userId);
        if (!membership) {
          return; // idempotent
        }
        assertCan(membership.role, "LEAVE_ROOM");
        
       

    await this.repo.removeMember(roomId, userId);
  }

  async update(roomId:string,userId:string,body:UpdateRoomBody) {
    const room = await this.repo.findById(roomId);
        if (!room) throw new NotFoundError("Room", roomId);
    
        const membership = await this.repo.findMembership(roomId, userId);
        if (!membership || membership.role !== "HOST") {
          throw new ForbiddenError("Only the host can update room settings");
        }
    assertCan(membership.role, "UPDATE_ROOM");
        if (body.maxMembers !== undefined) {
              const count = await this.repo.countMembers(roomId);
              if (body.maxMembers < count) {
                throw new ValidationError(
                  `maxMembers cannot be less than current member count (${count})`,
                );

/** Simple per-socket rate limit: max N events per window. */
function createRateLimiter(max: number, windowMs: number) {
  const hits = new Map<string, number[]>();
  return (key: string): boolean => {
    const now = Date.now();
    const arr = (hits.get(key) ?? []).filter((t) => now - t < windowMs);
    if (arr.length >= max) {
      hits.set(key, arr);
      return false;
    }
    arr.push(now);
    hits.set(key, arr);
    return true;
  };
}

const chatRateOk = createRateLimiter(20, 10_000);   // 20 msgs / 10s
const reactionRateOk = createRateLimiter(30, 10_000);
const typingRateOk = createRateLimiter(10, 5_000);

import { logger } from "../config/logger.js";
import { roomService } from "../modules/room/room.container.js";
import { roomPresence } from "./room-presence.js";
import { socketRoomName } from "../modules/room/room.helper.js";
import type { Server } from "socket.io";
import type {
  ClientToServerEvents,
  ServerToClientEvents,
  InterServerEvents,
  SocketData,
} from "@syncstream/contracts";
import { config } from "../config/index.js";

type IoServer = Server<
  ClientToServerEvents,
  ServerToClientEvents,
  InterServerEvents,
  SocketData
>;


type Pending = {
  timer: NodeJS.Timeout;
  hostId: string;
};

class HostLifecycle {
  private readonly pending = new Map<string, Pending>(); // roomId â†’ pending

  /**
   * Call when host presence goes offline in a room.
   */
  onHostWentOffline(io: IoServer, roomId: string, hostId: string) {
    this.cancel(roomId); // reset if already pending

    const timer = setTimeout(() => {
      void this.expire(io, roomId, hostId);
    }, config.HOST_GRACE_MS);
    // Don't keep process alive solely for grace timers
    timer.unref?.();

    this.pending.set(roomId, { timer, hostId });
    logger.info("host grace started", { roomId, hostId, graceMs: config.HOST_GRACE_MS });
  }

  /**
   * Call when host presence becomes online again.
   */
  onHostCameOnline(roomId: string, hostId: string) {
    const p = this.pending.get(roomId);
    if (!p || p.hostId !== hostId) return;
    this.cancel(roomId);
    logger.info("host grace cancelled (reconnected)", { roomId, hostId });
  }

  cancel(roomId: string) {
    const p = this.pending.get(roomId);
    if (!p) return;
    clearTimeout(p.timer);
    this.pending.delete(roomId);
  }

  private async expire(io: IoServer, roomId: string, hostId: string) {
    this.pending.delete(roomId);

    // Host came back on another code path?
    if (roomPresence.isOnline(roomId, hostId)) {
      logger.info("host grace expired but host online; skip", { roomId, hostId });
      return;
    }

    try {
      const members = await roomService.buildMemberList(roomId);
      // Prefer online CO_HOST, then any online non-host member
      const candidate =
        members.find((m) => m.online && m.role === "CO_HOST" && m.userId !== hostId) ??
        members.find((m) => m.online && m.userId !== hostId);

      if (!candidate) {
        logger.warn("host grace expired; no eligible member", { roomId, hostId });
        // Optional: mark room inactive / delete â€” product choice
        return;
      }

      await roomService.forceTransferHost(roomId, hostId, candidate.userId);
      const updated = await roomService.buildMemberList(roomId);
      const channel = socketRoomName(roomId);

      io.to(channel).emit("room:host-changed", {
        roomId,
        previousHostId: hostId,
        newHostId: candidate.userId,
        reason: "DISCONNECT_GRACE",
      });
      io.to(channel).emit("room:members", { roomId, members: updated });

      logger.info("host transferred after grace", {
        roomId,
        previousHostId: hostId,
        newHostId: candidate.userId,
      });
    } catch (err) {
      logger.error("host grace transfer failed", { err, roomId, hostId });
    }
  }
}

export const hostLifecycle = new HostLifecycle();

        // Host leaving membership is already forbidden in service;
        // if somehow host, cancel grace rather than transfer
        if (wasHost) hostLifecycle.cancel(roomId);
      } else if (wentOffline) {
        // Still a member, but offline
        const members = await roomService.buildMemberList(roomId);
        io.to(channel).emit("room:members", { roomId, members });
        if (wasHost) {
          hostLifecycle.onHostWentOffline(io, roomId, user.id);
        }

        

        logger.info("room:leave ok", {
          roomId,
          userId: user.id,
          leaveMembership,
          wentOffline,
        });
      }
    } catch (err) {
      emitRoomError(
        socket,
        err,
        typeof raw === "object" && raw && "roomId" in raw
          ? String((raw as { roomId?: string }).roomId)
          : undefined,
      );
    }
  });

  socket.on("disconnect", () => {
    const changes = roomPresence.removeSocket(socket.id, user.id);
  
    for (const { roomId, userWentOffline } of changes) {
      if (!userWentOffline) continue;
  
      void (async () => {
        try {
          const members = await roomService.buildMemberList(roomId);
          io.to(socketRoomName(roomId)).emit("room:members", {
            roomId,
            members,
          });
  
          const hostId = await roomService.getRoomHostId(roomId);
          if (hostId === user.id) {
            hostLifecycle.onHostWentOffline(io, roomId, user.id);
          }
        } catch (err) {
          logger.warn(
            { err, roomId },
            "failed to broadcast presence on disconnect",
          );
        }
      })();
    }
  });
  socket.on("room:transfer-host", async (raw) => {
    let roomId: string | undefined;
    try {
      const parsed = TransferHostPayloadSchema.safeParse(raw);
      if (!parsed.success) {
        socket.emit("room:error", {
          code: "BAD_REQUEST",
          message: parsed.error.issues.map((i) => i.message).join(", "),
        });
        return;
      }
      roomId = parsed.data.roomId;
          const { toUserId } = parsed.data;
      const result = await roomService.transferHost(user.id, roomId, toUserId);
      const channel = socketRoomName(roomId);
  
      hostLifecycle.cancel(roomId); // voluntary transfer clears grace
  
      io.to(channel).emit("room:host-changed", {
        roomId,
        previousHostId: result.previousHostId,
        newHostId: result.newHostId,
        reason: "TRANSFER",
      });
      io.to(channel).emit("room:members", { roomId, members: result.members });
    } catch (err) {
      emitRoomError(socket, err, roomId);

            });
            return;
          }
          roomId = parsed.data.roomId;

      // Optional: track userId â†’ socketIds if you need hard kick
    } catch (err) {
      emitRoomError(socket, err);
    }
  });

    } catch (err) {
      emitRoomError(socket, err);

  });

/**
 * Single-process presence:
 * roomId â†’ userId â†’ Set<socketId>
 *
 * Online = at least one socket for that user in the room.
 */
class RoomPresence {
  private readonly rooms = new Map<string, Map<string, Set<string>>>();

  /** socketId â†’ roomIds (for disconnect cleanup) */
  private readonly socketRooms = new Map<string, Set<string>>();

  add(roomId: string, userId: string, socketId: string): void {
    let users = this.rooms.get(roomId);
    if (!users) {
      users = new Map();
      this.rooms.set(roomId, users);
    }
    let sockets = users.get(userId);
    if (!sockets) {
      sockets = new Set();
      users.set(userId, sockets);
    }
    sockets.add(socketId);

    let sr = this.socketRooms.get(socketId);
    if (!sr) {
      sr = new Set();
      this.socketRooms.set(socketId, sr);
    }
    sr.add(roomId);
  }

  /**
   * Remove one socket from a room.
   * @returns true if the user became offline in that room
   */
  remove(roomId: string, userId: string, socketId: string): boolean {
    const users = this.rooms.get(roomId);
    if (!users) return false;

    const sockets = users.get(userId);
    if (!sockets) return false;

    sockets.delete(socketId);
    if (sockets.size === 0) {
      users.delete(userId);
      if (users.size === 0) this.rooms.delete(roomId);
      this.unlinkSocketRoom(socketId, roomId);
      return true;
    }
    this.unlinkSocketRoom(socketId, roomId);
    return false;
  }

  /** On disconnect: drop socket from all rooms. Returns rooms where user went offline. */
  removeSocket(
    socketId: string,
    userId: string,
  ): Array<{ roomId: string; userWentOffline: boolean }> {
    const roomIds = this.socketRooms.get(socketId);
    if (!roomIds || roomIds.size === 0) return [];

    const result: Array<{ roomId: string; userWentOffline: boolean }> = [];
    for (const roomId of [...roomIds]) {
      const userWentOffline = this.remove(roomId, userId, socketId);
      result.push({ roomId, userWentOffline });
    }
    this.socketRooms.delete(socketId);
    return result;
  }

  isOnline(roomId: string, userId: string): boolean {
    const sockets = this.rooms.get(roomId)?.get(userId);
    return Boolean(sockets && sockets.size > 0);
  }

  getOnlineUserIds(roomId: string): Set<string> {
    const users = this.rooms.get(roomId);
    if (!users) return new Set();
    return new Set(users.keys());
  }

  private unlinkSocketRoom(socketId: string, roomId: string) {
    const sr = this.socketRooms.get(socketId);
    if (!sr) return;
    sr.delete(roomId);
    if (sr.size === 0) this.socketRooms.delete(socketId);
  }
}

export const roomPresence = new RoomPresence();

import { Server, Socket } from "socket.io";
import { verifyAccessToken } from "../modules/auth/auth.helper.js";
import { authService } from "../modules/auth/auth.container.js";
import { SafeUser } from "../modules/auth/auth.types.js";
import { logger } from "../config/logger.js";
/**
 * Handshake auth: client must send
 *   io(url, { auth: { token: accessToken } })
 * Do NOT put the token in the query string.
 */
export async function verifySocketIdentity(socket: Socket): Promise<SafeUser> {
  const token =
    (socket.handshake.auth as { token?: string } | undefined)?.token ??
    extractBearer(socket.handshake.headers.authorization);
  if (!token) {
    throw new Error("Unauthorized");
  }

  const payload = verifyAccessToken(token);
  return authService.resolveUser(payload.sub);
}

function extractBearer(header: string | string[] | undefined): string | null {
  if (!header || Array.isArray(header)) return null;
  if (!header.startsWith("Bearer ")) return null;
  return header.slice(7).trim() || null;
}

export function registerSocketAuth(io: Server) {
  io.use(async (socket, next) => {
    try {
      const user = await verifySocketIdentity(socket);
      socket.data.user = user;
      next();
    } catch (err) {
      logger.warn("Socket auth failed", {
        socketId: socket.id,
        error: err instanceof Error ? err.message : String(err),
      });

      next(new Error("Unauthorized"));
    }
  });
}

import type { Server as HttpServer } from "node:http";

  SocketData,
} from "@syncstream/contracts";

import { logger } from "../config/logger.js";
import { registerSocketAuth } from "./socket-auth.js";
import { registerRoomHandlers } from "./room-handlers.js";
import { registerPlaybackHandlers } from "./playback-handlers.js";
import { registerChatHandlers } from "./chat-handlers.js";

export function createSocketServer(httpServer: HttpServer) {
  const io = new Server<
    ClientToServerEvents,
    ServerToClientEvents,
    InterServerEvents,
    SocketData
  >(httpServer, {
    path: config.SOCKET_PATH,
    cors: {
      origin: config.CLIENT_ORIGINS,
      credentials: true,
    },
  });

  registerSocketAuth(io);

  io.on("connection", (socket) => {
    const user = socket.data.user;
    logger.info("Socket connected", {
      socketId: socket.id,
      userId: user?.id,
    });

    socket.emit("server:hello", {
      message: `Connected as ${user?.email ?? "unknown"}`,
      serverTime: Date.now(),
      socketId: socket.id,
    });

    socket.on("client:ping", (raw) => {
      const parsed = ClientPingSchema.safeParse(raw);
      if (!parsed.success) {
        logger.warn("Invalid client:ping", { socketId: socket.id });
        return;
      }
      socket.emit("server:pong", {
        serverTime: Date.now(),
        clientTime: parsed.data.clientTime,
      });
    });

    registerRoomHandlers(io, socket);
    registerPlaybackHandlers(io, socket);  
   registerChatHandlers(io, socket); 

    socket.on("disconnect", (reason) => {
      logger.info("Socket disconnected", {
        socketId: socket.id,
        userId: user?.id,
        reason,
      });
    });
  });

  return io;
}

import type { Request, Response, NextFunction, RequestHandler } from "express";

export function asyncHandler(
  fn: (req: Request, res: Response, next: NextFunction) => Promise<unknown>,
): RequestHandler {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

export class AppError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly statusCode: number = 500,
    public readonly details?: unknown,
  ) {
    super(message);
    this.name = "AppError";
    Object.setPrototypeOf(this, new.target.prototype);
  }
}

export class NotFoundError extends AppError {
  constructor(resource: string, id?: string) {
    super(
      "NOT_FOUND",
      id ? `${resource} with id ${id} not found` : `${resource} not found`,
      404,
    );
  }
}

export class BadRequestError extends AppError {
  constructor(message = "Bad Request", details?: unknown) {
    super("BAD_REQUEST", message, 400, details);
  }
}

export class ValidationError extends AppError {
  constructor(message: string, details?: unknown) {
    super("VALIDATION_ERROR", message, 400, details);
  }
}

export class UnauthorizedError extends AppError {
  constructor(message = "Unauthorized", details?: unknown) {
    super("UNAUTHORIZED", message, 401, details);
  }
}

export class ConflictError extends AppError {
  constructor(message: string, details?: unknown) {
    super("CONFLICT", message, 409, details);
  }
}

  constructor(message = "Forbidden", details?: unknown) {
    super("FORBIDDEN", message, 403, details);
  }
}
