import express from "express";
import { getReadyStatus } from "./infrastructure/ready.js";
import { metrics } from "./infrastructure/metrics.js";
import { corsMiddleware } from "./middlewares/cors.middleware.js";
import { requestLogger } from "./middlewares/req.middleware.js";
import { errorMiddleware } from "./middlewares/error.middleware.js";
import { requestIdMiddleware } from "./middlewares/request-id.middleware.js";
import {
  authRateLimit,
  defaultApiRateLimit,
} from "./middlewares/rate-limit.middleware.js";
import authRoutes from "./modules/auth/auth.route.js";
import roomRoutes from "./modules/room/room.route.js";

export function createApp() {
  const app = express();

  app.use(requestIdMiddleware);
  app.use(corsMiddleware);
  app.use(express.json({ limit: "64kb" }));
  app.use(requestLogger);
  app.use(defaultApiRateLimit);

  app.get("/health", (_req, res) => {
    res.status(200).json({
      status: "ok",
      service: "syncstream-api",
      timestamp: new Date().toISOString(),
    });
  });

  app.get("/ready", async (_req, res) => {
    const result = await getReadyStatus();
    res.status(result.ready ? 200 : 503).json({
      status: result.ready ? "ready" : "not_ready",
      service: "syncstream-api",
      checks: result.checks,
      timestamp: new Date().toISOString(),
    });
  });

  app.get("/metrics", (_req, res) => {
    res.status(200).json({
      service: "syncstream-api",
      timestamp: new Date().toISOString(),
      metrics: metrics.snapshot(),
    });
  });

  app.use("/auth", authRateLimit, authRoutes);
  app.use("/rooms", roomRoutes);

  app.use(errorMiddleware);
  return app;
}

import { createServer } from "node:http";
import { createApp } from "./app.js";
import { config } from "./config/index.js";
import { logger } from "./config/logger.js";
import { prisma } from "./infrastructure/prisma.js";
import { closeRedis } from "./infrastructure/redis.js";
import { createSocketServer } from "./modules/realtime/socket-server.js";

async function main() {
  const app = createApp();
  const httpServer = createServer(app);

  const io = createSocketServer(httpServer);

  httpServer.listen(config.PORT, () => {
    logger.info(`syncstream-api listening on port ${config.PORT}`);
    logger.info(`Socket.IO path: ${config.SOCKET_PATH}`);
  });

  const shutdown = async (signal: string) => {
    logger.info(`Shutting down API (${signal})`);

    io.close();

    httpServer.close(async () => {
      try {
        await prisma.$disconnect();
        await closeRedis();
        logger.info("API shutdown complete");
        process.exit(0);
      } catch (err) {
        logger.error(err, "Error during API shutdown");
        process.exit(1);
      }
    });

    setTimeout(() => process.exit(1), 15_000).unref();
  };

  process.on("SIGINT", () => void shutdown("SIGINT"));
  process.on("SIGTERM", () => void shutdown("SIGTERM"));
}

main().catch((err) => {
  logger.error(err, "Failed to start server");
  process.exit(1);
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) return {};
  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }
  if (typeof meta === "object") return meta as Record<string, unknown>;
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    if (typeof msgOrMeta === "string") {
      baseLogger[level](toMergingObject(maybeMetaOrMsg), msgOrMeta);
      return;
    }
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;
    if (msg) baseLogger[level](obj, msg);
    else baseLogger[level](obj);
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import pg from "pg";
import { config } from "../config/index.js";
import { redis } from "./redis.js";

export type ReadyCheck = {
  name: string;
  ok: boolean;
  latencyMs?: number;
  error?: string;
};

export type ReadyResult = {
  ready: boolean;
  checks: ReadyCheck[];
};

async function checkPostgres(): Promise<ReadyCheck> {
  const start = Date.now();
  const pool = new pg.Pool({
    connectionString: config.DATABASE_URL,
    max: 1,
    connectionTimeoutMillis: 2_000,
  });
  try {
    await pool.query("SELECT 1");
    return { name: "postgres", ok: true, latencyMs: Date.now() - start };
  } catch (err) {
    return {
      name: "postgres",
      ok: false,
      latencyMs: Date.now() - start,
      error: err instanceof Error ? err.message : String(err),
    };
  } finally {
    await pool.end().catch(() => undefined);
  }
}

async function checkRedis(): Promise<ReadyCheck> {
  const start = Date.now();
  try {
    const pong = await redis.ping();
    return {
      name: "redis",
      ok: pong === "PONG",
      latencyMs: Date.now() - start,
    };
  } catch (err) {
    return {
      name: "redis",
      ok: false,
      latencyMs: Date.now() - start,
      error: err instanceof Error ? err.message : String(err),
    };
  }
}

export async function getReadyStatus(): Promise<ReadyResult> {
  const [pgCheck, redisCheck] = await Promise.all([
    checkPostgres(),
    checkRedis(),
  ]);
  return {
    ready: pgCheck.ok && redisCheck.ok,
    checks: [pgCheck, redisCheck],
  };
}

  NextFunction,

export const authenticate = async(
  req: Request,
  _res: Response,
  next: NextFunction,
): Promise<void> => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader) {
      return next(
        new UnauthorizedError("Authentication required"),
      );
    }

    if (!authHeader.startsWith("Bearer ")) {
      return next(
        new UnauthorizedError(
          "Invalid authentication header format",
        ),
      );
    }

    const accessToken = authHeader.split(" ")[1];

    if (!accessToken) {
      return next(
        new UnauthorizedError("Access token missing"),
      );
    }
    const payload = verifyAccessToken(accessToken);
  const user=await authService.resolveUser(payload.sub)

    req.user = user

    return next();
  } catch (error) {
    if (error instanceof jwt.TokenExpiredError) {
      return next(
        new UnauthorizedError("Access token expired"),
      );
    }

    if (error instanceof jwt.JsonWebTokenError) {
      return next(
        new UnauthorizedError("Invalid access token"),
      );
    }

    return next(error);
  }
};

import { AppError } from "@syncstream/shared-utils";
import { logger } from "../config/logger.js";
import { metrics } from "../infrastructure/metrics.js";

export function errorMiddleware(
  err: Error,
  req: Request,
  res: Response,
  _next: NextFunction,
): void {
  if (err instanceof AppError) {
    if (err.statusCode === 401) metrics.httpUnauthorized += 1;
    if (err.statusCode === 429) metrics.rateLimitHits += 1;
    logger.warn(
      { err, code: err.code, requestId: req.requestId, userId: req.user?.id },
      err.message,
    );
    res.status(err.statusCode).json({
      success: false,
      error: { code: err.code, message: err.message, details: err.details },
    });
    return;
  }

  metrics.unhandledErrors += 1;
  logger.error(
    { err, requestId: req.requestId, userId: req.user?.id },
    "Unhandled error",
  );
  res.status(500).json({
    success: false,
    error: {
      code: "INTERNAL_ERROR",
      message:
        process.env.NODE_ENV === "production"
          ? "An unexpected error occurred"
          : err.message,
    },
  });
}

// apps/api/src/middlewares/rate-limit.middleware.ts
import type { NextFunction, Request, Response } from "express";
import { AppError } from "@syncstream/shared-utils";
import { metrics } from "../infrastructure/metrics.js";

type Bucket = { timestamps: number[] };

export function createRateLimit(options: {
  windowMs: number;
  max: number;
  keyFn?: (req: Request) => string;
  code?: string;
  message?: string;
}) {
  const buckets = new Map<string, Bucket>();
  const {
    windowMs,
    max,
    keyFn = (req) => req.ip ?? "unknown",
    code = "RATE_LIMITED",
    message = "Too many requests; slow down",
  } = options;

  setInterval(() => {
    const now = Date.now();
    for (const [k, b] of buckets) {
      b.timestamps = b.timestamps.filter((t) => now - t < windowMs);
      if (b.timestamps.length === 0) buckets.delete(k);
    }
  }, Math.max(windowMs, 30_000)).unref?.();

  return (req: Request, _res: Response, next: NextFunction): void => {
    const key = keyFn(req);
    const now = Date.now();
    const bucket = buckets.get(key) ?? { timestamps: [] };
    bucket.timestamps = bucket.timestamps.filter((t) => now - t < windowMs);

    if (bucket.timestamps.length >= max) {
      metrics.rateLimitHits += 1;
      buckets.set(key, bucket);
      next(new AppError(code, message, 429));
      return;
    }

    bucket.timestamps.push(now);
    buckets.set(key, bucket);
    next();
  };
}

export const defaultApiRateLimit = createRateLimit({
  windowMs: 60_000,
  max: 120,
});

export const authRateLimit = createRateLimit({
  windowMs: 60_000,
  max: 20,
  message: "Too many auth attempts; try again later",
});

type ValidationTarget = "body" | "params" | "query";

export const validate =
  <T>(schema: ZodType<T>, target: ValidationTarget = "body") =>
  (req: Request, res: Response, next: NextFunction): void => {
    const result = schema.safeParse(req[target]);
    if (!result.success) {
      const message = result.error.issues
        .map(({ path, message }) => `${path.join(".")}:${message}`)
        .join(",");
      return next(new BadRequestError(message));
    }
    switch (target) {
      case "body":
        req.body = result.data;
        break;

      case "params":
        req.params = result.data as ParamsDictionary;
        break;

      case "query":
        res.locals.validatedQuery = result.data;
        break;
    }

    next();
  };

import jwt, { SignOptions } from "jsonwebtoken";
import { config } from "../../config/index.js";
import crypto from "node:crypto";
import { UnauthorizedError } from "@syncstream/shared-utils";

export type AccessPayload = {
  sub: string;
  typ: "access";
};

export function signAccessToken(userId: string): string {
  return jwt.sign(
    { sub: userId, typ: "access" } satisfies AccessPayload,
    config.JWT_ACCESS_SECRET,
    { expiresIn: config.JWT_ACCESS_EXPIRES_IN as SignOptions["expiresIn"] },
  );
}


export function verifyAccessToken(token: string): AccessPayload {
  try {
    const payload = jwt.verify(token, config.JWT_ACCESS_SECRET) as AccessPayload;
    if (payload.typ !== "access" || !payload.sub) {
      throw new UnauthorizedError("Invalid access token");
    }
    return payload;
  } catch {
    throw new UnauthorizedError("Invalid or expired access token");
  }
}

/** Opaque refresh token (not a JWT) â€” store only the hash. */
export function generateRefreshToken(): string {
  return crypto.randomBytes(48).toString("base64url");
}

export function hashToken(token: string): string {
  return crypto.createHash("sha256").update(token).digest("hex");
}

export function refreshExpiresAt(): Date {
  const match = /^(\d+)([dhms])$/.exec(config.JWT_REFRESH_EXPIRES_IN);
  const now = Date.now();
  if (!match) {
    return new Date(now + 7 * 24 * 60 * 60 * 1000);
  }
  const n = Number(match[1]);
  const unit = match[2];
  const ms =
    unit === "d" ? n * 86400000 :
    unit === "h" ? n * 3600000 :
    unit === "m" ? n * 60000 :
    n * 1000;
  return new Date(now + ms);
}

import {

  UnauthorizedError,
  ValidationError,
} from "@syncstream/shared-utils";
import { AuthInterface } from "./auth.interface.js";
import { LoginBody, RegisterBody } from "./auth.schema.js";
import bcrypt from "bcryptjs";
import { config } from "../../config/index.js";
import {
  generateRefreshToken,
  hashToken,
  refreshExpiresAt,
  signAccessToken,
} from "./auth.helper.js";
import { SafeUser, TokenPair } from "./auth.types.js";

export class AuthService {
  constructor(private readonly repo: AuthInterface) {}

  async register(input: RegisterBody) {
    const { email, name, password } = input;
    const existing = await this.repo.findByEmail(email);
    if (existing) {
      throw new ConflictError("Email already registered");
    }
    const passwordHash = await bcrypt.hash(password, config.BCRYPT_ROUNDS);
    const user = await this.repo.createUser({
      name,
      email,
      passwordHash,
    });
    const tokens = await this.issueTokens(user.id);
    return { user, tokens };
  }
  async login(input: LoginBody) {
    const user = await this.repo.findByEmail(input.email);
    if (!user) {
      throw new UnauthorizedError("Invalid email or password");
    }
    const isPasswordValid = await bcrypt.compare(
      input.password,
      user.passwordHash,
    );

    if (!isPasswordValid) {
      throw new UnauthorizedError("Invalid email or password");
    }

    const tokens = await this.issueTokens(user.id);
    return {
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        avatarUrl: user.avatarUrl,
      },
      tokens,
    };
  }

  async refresh(refreshToken: string): Promise<TokenPair> {
    if (!refreshToken) {
      throw new ValidationError("refreshToken is required");
    }

    const tokenHash = hashToken(refreshToken);
    const token = await this.repo.findValidRefreshToken(tokenHash);
    if (!token) {
      throw new UnauthorizedError("Invalid or expired refresh token");
    }

    // Rotate: revoke old, issue new pair
    await this.repo.revokeRefreshToken(token.id);
    return this.issueTokens(token.userId);
  }

  async logout(refreshToken: string | undefined): Promise<void> {
    if (!refreshToken) return;
    const tokenHash = hashToken(refreshToken);
    const row = await this.repo.findValidRefreshToken(tokenHash);
    if (row) {
      await this.repo.revokeRefreshToken(row.id);
    }
  }

  async logoutAll(userId: string): Promise<void> {
    await this.repo.revokeAllForUser(userId);
  }

  async getMe(userId: string): Promise<SafeUser> {
    const user = await this.repo.findById(userId);
    if (!user) {
      throw new UnauthorizedError("User not found");
    }
    return user;
  }

  /** Used by HTTP + Socket middleware after JWT verify. */
  async resolveUser(userId: string): Promise<SafeUser> {
    return this.getMe(userId);
  }
  private async issueTokens(userId: string): Promise<TokenPair> {
    const accessToken = signAccessToken(userId);
    const refreshToken = generateRefreshToken();
    const tokenHash = hashToken(refreshToken);

    await this.repo.createRefreshToken({
      userId,
      tokenHash,
      expiresAt: refreshExpiresAt(),
    });

    return {
      accessToken,
      refreshToken,
      expiresIn: config.JWT_ACCESS_EXPIRES_IN,
    };
  }
}

import {
  PlaybackCommand,
  PlaybackState,
  RoomRole,
} from "@syncstream/contracts";
import { ValidationError } from "@syncstream/shared-utils";
import { assertCan } from "../room/room.permissions.js";
import { redis } from "../../infrastructure/redis.js";
import { logger } from "../../config/logger.js";

const MAX_RECENT_COMMANDS = 64;

const STATE_KEY = (roomId: string) => `playback:room:${roomId}`;
const CMDS_KEY = (roomId: string) => `playback:cmds:${roomId}`;

/**
 * Atomic apply-command Lua script.
 *
 * KEYS[1] = state key
 * KEYS[2] = recent-commands set
 * ARGV[1] = commandId
 * ARGV[2] = new state JSON
 * ARGV[3] = max recent commands
 *
 * Returns JSON:
 *   { "duplicate": true,  "state": <current|null> }
 *   { "duplicate": false, "state": <new> }
 */
const APPLY_SCRIPT = `
local stateKey = KEYS[1]
local cmdsKey  = KEYS[2]
local cmdId    = ARGV[1]
local newJson  = ARGV[2]
local maxCmds  = tonumber(ARGV[3])

if redis.call("SISMEMBER", cmdsKey, cmdId) == 1 then
  local current = redis.call("GET", stateKey)
  return cjson.encode({
    duplicate = true,
    state     = current and cjson.decode(current) or cjson.null
  })
end

redis.call("SET", stateKey, newJson)
redis.call("SADD", cmdsKey, cmdId)

local size = redis.call("SCARD", cmdsKey)
if size > maxCmds then
  local members = redis.call("SRANDMEMBER", cmdsKey, size - maxCmds)
  if type(members) == "table" then
    for _, m in ipairs(members) do
      redis.call("SREM", cmdsKey, m)
    end
  end
end

return cjson.encode({
  duplicate = false,
  state     = cjson.decode(newJson)
})
`;

export class PlaybackService {
  /** Current state or null if no video has been selected yet. */
  async getState(roomId: string): Promise<PlaybackState | null> {
    const raw = await redis.get(STATE_KEY(roomId));
    if (!raw) return null;
    try {
      return JSON.parse(raw) as PlaybackState;
    } catch {
      logger.error({ roomId }, "Corrupt playback state in Redis");
      return null;
    }
  }

  /**
   * Apply a host/co-host command.
   * Throws AppError subclasses on validation / permission failure.
   */
  async applyCommand(
    roomId: string,
    _userId: string,
    role: RoomRole,
    cmd: PlaybackCommand,
  ): Promise<{ state: PlaybackState; duplicate: boolean }> {
    // 1. Permission
    assertCan(role, "PLAYBACK_CONTROL");

    // 2. Load current state for transition logic
    const current = await this.getState(roomId);
    const now = Date.now();

    // 3. Compute next state (pure)
    let next: PlaybackState;

    switch (cmd.action) {
      case "CHANGE_VIDEO": {
        if (!cmd.videoId) {
          throw new ValidationError("videoId is required for CHANGE_VIDEO");
        }
        next = {
          videoId: cmd.videoId,
          status: "PAUSED",
          position: 0,
          updatedAt: now,
          playbackRate: 1,
          sequence: current ? current.sequence + 1 : 1,
        };
        break;
      }
      case "PLAY": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        const position =
          cmd.position !== undefined
            ? cmd.position
            : this.expectedPosition(current, now);
        next = {
          ...current,
          status: "PLAYING",
          position,
          updatedAt: now,
          sequence: current.sequence + 1,
        };
        break;
      }
      case "PAUSE": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        const position = this.expectedPosition(current, now);
        next = {
          ...current,
          status: "PAUSED",
          position,
          updatedAt: now,
          sequence: current.sequence + 1,
        };
        break;
      }
      case "SEEK": {
        if (!current) {
          throw new ValidationError("No video selected â€“ CHANGE_VIDEO first");
        }
        if (cmd.position === undefined) {
          throw new ValidationError("position is required for SEEK");
        }
        if (cmd.position < 0) {
          throw new ValidationError("position cannot be negative");
        }
        next = {
          ...current,
          position: cmd.position,
          updatedAt: now,
          sequence: current.sequence + 1,
        };
        break;
      }
      default: {
        const _exhaustive: never = cmd.action;
        throw new ValidationError(`Unknown action: ${_exhaustive}`);
      }
    }

    // 4. Atomic commit via Lua
    const resultJson = (await redis.eval(
      APPLY_SCRIPT,
      2,
      STATE_KEY(roomId),
      CMDS_KEY(roomId),
      cmd.commandId,
      JSON.stringify(next),
      String(MAX_RECENT_COMMANDS),
    )) as string;

    const result = JSON.parse(resultJson) as {
      duplicate: boolean;
      state: PlaybackState | null;
    };

    if (result.duplicate) {
      if (!result.state) {
        throw new ValidationError("Duplicate command but no state exists");
      }
      return { state: result.state, duplicate: true };
    }

    return { state: result.state as PlaybackState, duplicate: false };
  }

  /** Called when a room is deleted. */
  async clearRoom(roomId: string): Promise<void> {
    await redis.del(STATE_KEY(roomId), CMDS_KEY(roomId));
  }

  /**
   * Expected playback position at time `now`.
   */
  expectedPosition(state: PlaybackState, now: number): number {
    if (state.status === "PAUSED") return state.position;
    const elapsedSec = (now - state.updatedAt) / 1000;
    return Math.max(0, state.position + state.playbackRate * elapsedSec);
  }
}

import { config } from "../../config/index.js";

type Pending = {
  timer: NodeJS.Timeout;
  hostId: string;
};

class HostLifecycle {
  private readonly pending = new Map<string, Pending>(); // roomId â†’ pending

  /**
   * Call when host presence goes offline in a room.
   */
  onHostWentOffline(io: IoServer, roomId: string, hostId: string) {
    this.cancel(roomId); // reset if already pending

    const timer = setTimeout(() => {
      void this.expire(io, roomId, hostId);
    }, config.HOST_GRACE_MS);
    // Don't keep process alive solely for grace timers
    timer.unref?.();

    this.pending.set(roomId, { timer, hostId });
    logger.info("host grace started", { roomId, hostId, graceMs: config.HOST_GRACE_MS });
  }

  /**
   * Call when host presence becomes online again.
   */
  onHostCameOnline(roomId: string, hostId: string) {
    const p = this.pending.get(roomId);
    if (!p || p.hostId !== hostId) return;
    this.cancel(roomId);
    logger.info("host grace cancelled (reconnected)", { roomId, hostId });
  }

  cancel(roomId: string) {
    const p = this.pending.get(roomId);
    if (!p) return;
    clearTimeout(p.timer);
    this.pending.delete(roomId);
  }

  private async expire(io: IoServer, roomId: string, hostId: string) {
    this.pending.delete(roomId);

    // Host came back on another code path?
    if (await roomPresence.isOnline(roomId, hostId)) {
      logger.info("host grace expired but host online; skip", { roomId, hostId });
      return;
    }

    try {
      const members = await roomService.buildMemberList(roomId);
      // Prefer online CO_HOST, then any online non-host member
      const candidate =
        members.find((m) => m.online && m.role === "CO_HOST" && m.userId !== hostId) ??
        members.find((m) => m.online && m.userId !== hostId);

      if (!candidate) {
        logger.warn("host grace expired; no eligible member", { roomId, hostId });
        // Optional: mark room inactive / delete â€” product choice
        return;
      }

      await roomService.forceTransferHost(roomId, hostId, candidate.userId);
      const updated = await roomService.buildMemberList(roomId);
      const channel = socketRoomName(roomId);

      io.to(channel).emit("room:host-changed", {
        roomId,
        previousHostId: hostId,
        newHostId: candidate.userId,
        reason: "DISCONNECT_GRACE",
      });
      io.to(channel).emit("room:members", { roomId, members: updated });

      logger.info("host transferred after grace", {
        roomId,
        previousHostId: hostId,
        newHostId: candidate.userId,
      });
    } catch (err) {
      logger.error("host grace transfer failed", { err, roomId, hostId });
    }
  }
}

function emitPlaybackError(
  socket: AppSocket,
  err: unknown,
  roomId?: string,
  commandId?: string,
) {
  if (err instanceof AppError) {
    socket.emit("playback:error", {
      code: err.code,
      message: err.message,
      roomId,
      commandId,
    });
    return;
  }
  logger.error("playback handler error", { err });
  socket.emit("playback:error", {
    code: "INTERNAL_ERROR",
    message: "Unexpected playback error",
    roomId,
    commandId,
  });
}

export function registerPlaybackHandlers(io: IoServer, socket: AppSocket) {
  const user = socket.data.user;
  if (!user) return;

  // â”€â”€ Client asks for current state (reconnect / late join) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  socket.on("playback:sync", async (raw) => {
    try {
      const parsed = PlaybackSyncPayloadSchema.safeParse(raw);
      if (!parsed.success) {
        socket.emit("playback:error", {
          code: "BAD_REQUEST",
          message: parsed.error.issues.map((i) => i.message).join(", "),
        });
        return;
      }

      const { roomId } = parsed.data;

      // Must be a member
      await roomService.assertMember(roomId, user.id);

      const state = await playbackService.getState(roomId);

      if (!state) {
        // No video has been selected yet â€“ still a valid reply
        socket.emit("playback:snapshot", {
          roomId,
          state: {
            videoId: "",
            status: "PAUSED",
            position: 0,
            updatedAt: Date.now(),
            playbackRate: 1,
            sequence: 0,
          },
        });
        return;
      }

      socket.emit("playback:snapshot", { roomId, state });
    } catch (err) {
      emitPlaybackError(
        socket,
        err,
        typeof raw === "object" && raw && "roomId" in raw
          ? String((raw as { roomId?: string }).roomId)
          : undefined,
      );
    }
  });

  // â”€â”€ Host / co-host issues a command â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  socket.on("playback:command", async (raw) => {
    let commandId: string | undefined;
    let roomId: string | undefined;

    try {
      const parsed = PlaybackCommandSchema.safeParse(raw);
      if (!parsed.success) {
        socket.emit("playback:error", {
          code: "BAD_REQUEST",
          message: parsed.error.issues.map((i) => i.message).join(", "),
        });
        return;
      }

      const cmd = parsed.data;
      commandId = cmd.commandId;
      roomId = cmd.roomId;

      // Membership + role
      const membership = await roomService.assertMember(roomId, user.id);

      const result = await playbackService.applyCommand(
        roomId,
        user.id,
        membership.role,
        cmd,
      );

      // Even on duplicate we still broadcast the current state so the
      // client that retried gets the latest sequence.
      const channel = socketRoomName(roomId);
      io.to(channel).emit("playback:state", {
        roomId,
        state: result.state,
      });

      logger.info("playback:command accepted", {
        roomId,
        userId: user.id,
        action: cmd.action,
        sequence: result.state.sequence,
        duplicate: result.duplicate,
      });
    } catch (err) {
      emitPlaybackError(socket, err, roomId, commandId);
    }
  });
}

export function registerRoomHandlers(io: IoServer, socket: AppSocket) {
  const user = socket.data.user;
  if (!user) return;

  socket.on("room:join", async (raw) => {
    try {
      const parsed = RoomJoinPayloadSchema.safeParse(raw);
      if (!parsed.success) {
        socket.emit("room:error", {
          code: "BAD_REQUEST",
          message: parsed.error.issues.map((i) => i.message).join(", "),
        });
        return;
      }

      const result = await roomService.join(user.id, {
        roomId: parsed.data.roomId,
        code: parsed.data.code,
      });

      const roomId = result.room.id;
      const channel = socketRoomName(roomId);

      socket.join(channel);
      await roomPresence.add(roomId, user.id, socket.id);
      const room = result.room;
      if (room.hostId === user.id) {
        hostLifecycle.onHostCameOnline(roomId, user.id);
      }

      // Refresh members with presence
      const members = await roomService.buildMemberList(roomId);

      socket.emit("room:state", {
        room: result.room,
        members,
      });

      socket.emit("room:members", { roomId, members });
      const playbackState =await playbackService.getState(roomId);
            if (playbackState) {
              socket.emit("playback:snapshot", {
                roomId,
                state: playbackState,
              });
            }
      if (result.created) {
        const me = members.find((m) => m.userId === user.id);
        if (me) {
          socket.to(channel).emit("room:member-joined", {
            roomId,
            member: me,
          });
        }
        socket.to(channel).emit("room:members", { roomId, members });
      } else {
        // Reconnect / second tab: broadcast updated online flags
        socket.to(channel).emit("room:members", { roomId, members });
      }

      logger.info("room:join ok", {
        roomId,
        userId: user.id,
        socketId: socket.id,
        created: result.created,
      });
    } catch (err) {
      emitRoomError(socket, err, typeof raw === "object" && raw && "roomId" in raw
        ? String((raw as { roomId?: string }).roomId)
        : undefined);
    }
  });

  socket.on("room:leave", async (raw) => {
    try {
      const parsed = RoomLeavePayloadSchema.safeParse(raw);
      if (!parsed.success) {
        socket.emit("room:error", {
          code: "BAD_REQUEST",
          message: "Invalid leave payload",
        });
        return;
      }
  
      const { roomId, leaveMembership } = parsed.data;
      const channel = socketRoomName(roomId);
  
      const hostId = await roomService.getRoomHostId(roomId);
      const wasHost = hostId === user.id;
  
      socket.leave(channel);
      const wentOffline = await roomPresence.remove(roomId, user.id, socket.id);
  
      if (leaveMembership) {
        await roomService.leave(user.id, roomId);
        io.to(channel).emit("room:member-left", { roomId, userId: user.id });
        if (wasHost) hostLifecycle.cancel(roomId);
      } else if (wentOffline) {
        const members = await roomService.buildMemberList(roomId);
        io.to(channel).emit("room:members", { roomId, members });
        if (wasHost) {
          hostLifecycle.onHostWentOffline(io, roomId, user.id);
        }
      }
      // else: still online on another socket â€” no broadcast needed
  
      logger.info("room:leave ok", {
        roomId,
        userId: user.id,
        leaveMembership,
        wentOffline,
      });
    } catch (err) {
      emitRoomError(
        socket,
        err,
        typeof raw === "object" && raw && "roomId" in raw
          ? String((raw as { roomId?: string }).roomId)
          : undefined,
        
      );
    }
  });

  socket.on("disconnect", async() => {
    const changes = await roomPresence.removeSocket(socket.id, user.id);
  
    for (const { roomId, userWentOffline } of changes) {
      if (!userWentOffline) continue;

/**
 * Multi-instance presence backed by Redis.
 *
 * Key layout:
 *   presence:room:{roomId}          Hash  â†’ userId â†’ JSON({ sockets: string[], lastSeen: number })
 *   presence:socket:{socketId}      Set   â†’ roomIds this socket belongs to
 *
 * A user is considered online in a room while their entry still contains
 * at least one live socketId.
 */

import { redis } from "../../infrastructure/redis.js";
import { logger } from "../../config/logger.js";

const ROOM_KEY = (roomId: string) => `presence:room:${roomId}`;
const SOCKET_KEY = (socketId: string) => `presence:socket:${socketId}`;

type PresenceEntry = {
  sockets: string[];
  lastSeen: number;
};

class RoomPresence {
  async add(roomId: string, userId: string, socketId: string): Promise<void> {
    const key = ROOM_KEY(roomId);
    const raw = await redis.hget(key, userId);
    const data: PresenceEntry = raw
      ? JSON.parse(raw)
      : { sockets: [], lastSeen: Date.now() };

    if (!data.sockets.includes(socketId)) {
      data.sockets.push(socketId);
    }
    data.lastSeen = Date.now();

    await redis
      .multi()
      .hset(key, userId, JSON.stringify(data))
      .sadd(SOCKET_KEY(socketId), roomId)
      .exec();
  }

  /**
   * Remove one socket from a room.
   * @returns true if the user became offline in that room
   */
  async remove(
    roomId: string,
    userId: string,
    socketId: string,
  ): Promise<boolean> {
    const key = ROOM_KEY(roomId);
    const raw = await redis.hget(key, userId);
    if (!raw) return false;

    const data: PresenceEntry = JSON.parse(raw);
    data.sockets = data.sockets.filter((s) => s !== socketId);

    const pipeline = redis.multi();
    pipeline.srem(SOCKET_KEY(socketId), roomId);

    if (data.sockets.length === 0) {
      pipeline.hdel(key, userId);
      await pipeline.exec();
      return true; // user went offline
    }

    data.lastSeen = Date.now();
    pipeline.hset(key, userId, JSON.stringify(data));
    await pipeline.exec();
    return false;
  }

  /**
   * On disconnect: drop the socket from every room it was in.
   * Returns the list of rooms where the user became offline.
   */
  async removeSocket(
    socketId: string,
    userId: string,
  ): Promise<Array<{ roomId: string; userWentOffline: boolean }>> {
    const roomIds = await redis.smembers(SOCKET_KEY(socketId));
    if (roomIds.length === 0) return [];

    const result: Array<{ roomId: string; userWentOffline: boolean }> = [];

    for (const roomId of roomIds) {
      const userWentOffline = await this.remove(roomId, userId, socketId);
      result.push({ roomId, userWentOffline });
    }

    // clean up the socket key itself
    await redis.del(SOCKET_KEY(socketId));
    return result;
  }

  async isOnline(roomId: string, userId: string): Promise<boolean> {
    const raw = await redis.hget(ROOM_KEY(roomId), userId);
    if (!raw) return false;
    const data: PresenceEntry = JSON.parse(raw);
    return data.sockets.length > 0;
  }

  async getOnlineUserIds(roomId: string): Promise<Set<string>> {
    const all = await redis.hgetall(ROOM_KEY(roomId));
    return new Set(Object.keys(all));
  }

  /** Debug helper â€“ not used in production paths */
  async debugRoom(roomId: string): Promise<Record<string, PresenceEntry>> {
    const all = await redis.hgetall(ROOM_KEY(roomId));
    const out: Record<string, PresenceEntry> = {};
    for (const [uid, raw] of Object.entries(all)) {
      out[uid] = JSON.parse(raw);
    }
    return out;
  }
}

export const roomPresence = new RoomPresence();

import { Server, Socket } from "socket.io";
import { verifyAccessToken } from "../auth/auth.helper.js";
import { authService } from "../auth/auth.container.js";
import { SafeUser } from "../auth/auth.types.js";
import { logger } from "../../config/logger.js";
import { metrics } from "../../infrastructure/metrics.js";
/**
 * Handshake auth: client must send
 *   io(url, { auth: { token: accessToken } })
 * Do NOT put the token in the query string.
 */
export async function verifySocketIdentity(socket: Socket): Promise<SafeUser> {
  const token =
    (socket.handshake.auth as { token?: string } | undefined)?.token ??
    extractBearer(socket.handshake.headers.authorization);
  if (!token) {
    throw new Error("Unauthorized");
  }

  const payload = verifyAccessToken(token);
  return authService.resolveUser(payload.sub);
}

function extractBearer(header: string | string[] | undefined): string | null {
  if (!header || Array.isArray(header)) return null;
  if (!header.startsWith("Bearer ")) return null;
  return header.slice(7).trim() || null;
}

export function registerSocketAuth(io: Server) {
  io.use(async (socket, next) => {
    try {
      const user = await verifySocketIdentity(socket);
      socket.data.user = user;
      next();
    } catch (err) {
      metrics.socketUnauthorized += 1;
      logger.warn("Socket auth failed", {
        socketId: socket.id,
        error: err instanceof Error ? err.message : String(err),
      });

      next(new Error("Unauthorized"));
    }
  });
}

export function createSocketServer(httpServer: HttpServer) {
  const pubClient = redis.duplicate();
    const subClient = redis.duplicate();
  const io = new Server<
    ClientToServerEvents,
    ServerToClientEvents,
    InterServerEvents,
    SocketData
  >(httpServer, {
    path: config.SOCKET_PATH,
    cors: {
      origin: config.CLIENT_ORIGINS,
      credentials: true,
    
    },
    adapter: createAdapter(pubClient, subClient),
  });

  registerSocketAuth(io);

  io.on("connection", (socket) => {
    const user = socket.data.user;
    metrics.incSockets(1);
    logger.info("Socket connected", {
      socketId: socket.id,
      userId: user?.id,
    });

    socket.emit("server:hello", {
      message: `Connected as ${user?.email ?? "unknown"}`,
      serverTime: Date.now(),
      socketId: socket.id,
    });

    socket.on("client:ping", (raw) => {
      const parsed = ClientPingSchema.safeParse(raw);
      if (!parsed.success) {
        logger.warn("Invalid client:ping", { socketId: socket.id });
        return;
      }
      socket.emit("server:pong", {
        serverTime: Date.now(),
        clientTime: parsed.data.clientTime,
      });
    });

    registerRoomHandlers(io, socket);
    registerPlaybackHandlers(io, socket);  
   registerChatHandlers(io, socket); 

    socket.on("disconnect", (reason) => {
      metrics.incSockets(-1);
      logger.info("Socket disconnected", {
        socketId: socket.id,
        userId: user?.id,
        reason,
      });
    });
  });

  return io;
}

/** Unambiguous alphabet (no 0/O, 1/I/L). ~8 chars â‰ˆ 40 bits. */
const ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ";

export function generateRoomCode(length = 8):string {
  const bytes = crypto.randomBytes(length)
  let code = "";
    for (let i = 0; i < length; i++) {
      code += ALPHABET[bytes[i]! % ALPHABET.length];
    }
    return code;
}

import type { RoomRole } from "@syncstream/contracts";
import { ForbiddenError } from "@syncstream/shared-utils";

export type RoomAction =
  | "PLAYBACK_CONTROL"   // play / pause / seek / change video
  | "SEND_CHAT"
  | "SEND_REACTION"
  | "UPDATE_ROOM"
  | "DELETE_ROOM"
  | "REMOVE_MEMBER"
  | "CHANGE_ROLE"        // promote / demote
  | "TRANSFER_HOST"
  | "LEAVE_ROOM";


  const MATRIX: Record<RoomAction, readonly RoomRole[]> = {
    PLAYBACK_CONTROL: ["HOST", "CO_HOST"],
    SEND_CHAT: ["HOST", "CO_HOST", "VIEWER"],
    SEND_REACTION: ["HOST", "CO_HOST", "VIEWER"],
    UPDATE_ROOM: ["HOST"],
    DELETE_ROOM: ["HOST"],
    REMOVE_MEMBER: ["HOST"],
    CHANGE_ROLE: ["HOST"],
    TRANSFER_HOST: ["HOST"],
    LEAVE_ROOM: ["CO_HOST", "VIEWER"], // host must transfer or delete
};

  export function can(role: RoomRole, action: RoomAction): boolean {
    return MATRIX[action].includes(role);
  }
  
  export function assertCan(role: RoomRole, action: RoomAction): void {
    if (!can(role, action)) {
      throw new ForbiddenError(`Role ${role} cannot perform ${action}`);
    }
  }

  NotFoundError,
} from "@syncstream/shared-utils";
import { prisma } from "../../infrastructure/prisma.js";
import type { RoomInterface } from "./room.interface.js";
import type {
  CreateRoomInput,

  RoomRecord,
  UpdateRoomInput,
} from "./room.types.js";
import type { RoomRole } from "@syncstream/contracts";
import { generateRoomCode } from "./room.helper.js";

function toRoomRecord(row: {
  id: string;
  code: string;
  name: string;
  visibility: string;
  hostId: string;
  maxMembers: number;
  createdAt: Date;
  updatedAt: Date;
}): RoomRecord {
  return {
    id: row.id,
    code: row.code,
    name: row.name,
    visibility: row.visibility as RoomRecord["visibility"],
    hostId: row.hostId,
    maxMembers: row.maxMembers,
    createdAt: row.createdAt,
    updatedAt: row.updatedAt,
  };
}

export class RoomRepository implements RoomInterface {
  async createRoomWithHost(data: CreateRoomInput): Promise<RoomRecord> {
    const maxAttempts = 5;
    for (let attempt = 0; attempt < maxAttempts; attempt++) {
      const code = generateRoomCode(8);
      try {
        const room = await prisma.$transaction(async (tx) => {
          const created = await tx.room.create({
            data: {
              code,
              name: data.name,
              visibility: data.visibility ?? "PRIVATE",
              hostId: data.hostId,
              maxMembers: data.maxMembers ?? 10,
              members: {
                create: {
                  userId: data.hostId,
                  role: "HOST",
                },
              },
            },
          });
          return created;
        });
        return toRoomRecord(room);
      } catch (err: unknown) {
        // Unique violation on code â€” retry
        const codeConflict =
          typeof err === "object" &&
          err !== null &&
          "code" in err &&
          (err as { code?: string }).code === "P2002";
        if (codeConflict && attempt < maxAttempts - 1) continue;
        throw err;
      }
    }
    throw new ConflictError("Could not allocate a unique room code");
  }

  async findById(id: string): Promise<RoomRecord | null> {
    const row = await prisma.room.findUnique({ where: { id } });
    return row ? toRoomRecord(row) : null;
  }

  async findByCode(code: string): Promise<RoomRecord | null> {
    const row = await prisma.room.findUnique({
      where: { code: code.toUpperCase() },
    });
    return row ? toRoomRecord(row) : null;
  }

  async updateRoom(id: string, data: UpdateRoomInput): Promise<RoomRecord> {
    const row = await prisma.room.update({
      where: { id },
      data: {
        ...(data.name !== undefined ? { name: data.name } : {}),
        ...(data.visibility !== undefined ? { visibility: data.visibility } : {}),
        ...(data.maxMembers !== undefined ? { maxMembers: data.maxMembers } : {}),
      },
    });
    return toRoomRecord(row);
  }

  async deleteRoom(id: string): Promise<void> {
    await prisma.room.delete({ where: { id } });
  }

  async countMembers(roomId: string): Promise<number> {
    return prisma.roomMember.count({ where: { roomId } });
  }

  async findMembership(roomId: string, userId: string) {
    return prisma.roomMember.findUnique({
      where: { roomId_userId: { roomId, userId } },
      select: { roomId: true, userId: true, role: true, joinedAt: true },
    });
  }

  async listMembers(roomId: string): Promise<MemberRecord[]> {
    const rows = await prisma.roomMember.findMany({
      where: { roomId },
      orderBy: { joinedAt: "asc" },
      include: {
        user: {
          select: { id: true, name: true, avatarUrl: true },
        },
      },
    });
    return rows.map((r) => ({
      roomId: r.roomId,
      userId: r.userId,
      role: r.role as RoomRole,
      joinedAt: r.joinedAt,
      user: r.user,
    }));
  }

  async joinAsViewer(
    roomId: string,
    userId: string,
  ): Promise<{ membership: { role: RoomRole; joinedAt: Date }; created: boolean }> {
    return prisma.$transaction(async (tx) => {
      const existing = await tx.roomMember.findUnique({
        where: { roomId_userId: { roomId, userId } },
      });
      if (existing) {
        return {
          membership: {
            role: existing.role as RoomRole,
            joinedAt: existing.joinedAt,
          },
          created: false,
        };
      }

      const room = await tx.room.findUnique({ where: { id: roomId } });
      if (!room) {
        throw new NotFoundError("Room", roomId);
      }

      const count = await tx.roomMember.count({ where: { roomId } });
      if (count >= room.maxMembers) {
        throw new ConflictError("Room is full");
      }

      const created = await tx.roomMember.create({
        data: {
          roomId,
          userId,
          role: "VIEWER",
        },
      });

      return {
        membership: {
          role: created.role as RoomRole,
          joinedAt: created.joinedAt,
        },
        created: true,
      };
    });
  }

  async removeMember(roomId: string, userId: string): Promise<void> {
    await prisma.roomMember.delete({
      where: { roomId_userId: { roomId, userId } },
    });
  }

  async setMemberRole(roomId: string, userId: string, role: RoomRole): Promise<void> {
    await prisma.roomMember.update({
      where: { roomId_userId: { roomId, userId } },
      data:{role}
    })
  }
  async transferHost(roomId: string, fromUserId: string, toUserId: string): Promise<void> {
    await prisma.$transaction(async (tx) => {
      const room = await tx.room.findUnique({ where: { id: roomId } });
          if (!room) throw new NotFoundError("Room", roomId);
          if (room.hostId !== fromUserId) {
            throw new ForbiddenError("Only the current host can transfer");
          }
      
          const target = await tx.roomMember.findUnique({
            where: { roomId_userId: { roomId, userId: toUserId } },
          });
          if (!target) {
            throw new NotFoundError("Member", toUserId);
      }
      await tx.roomMember.update({
            where: { roomId_userId: { roomId, userId: fromUserId } },
            data: { role: "VIEWER" },
          });
          await tx.roomMember.update({
            where: { roomId_userId: { roomId, userId: toUserId } },
            data: { role: "HOST" },
          });
          await tx.room.update({
            where: { id: roomId },
            data: { hostId: toUserId },
          });
    })
  }
}

"use client";

import { FormEvent, useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { useAuth } from "@/context/AuthContext";
import { ApiError, roomsApi } from "@/lib/api";
import {
  Alert,
  Button,
  Field,
  Header,
  Input,
  Shell,
} from "@/components/ui";

export default function RoomsLobbyPage() {
  const { user, loading, logout } = useAuth();
  const router = useRouter();
  const [name, setName] = useState("Movie night");
  const [code, setCode] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    if (!loading && !user) router.replace("/login");
  }, [loading, user, router]);

  async function createRoom(e: FormEvent) {
    e.preventDefault();
    setBusy(true);
    setError(null);
    try {
      const room = await roomsApi.create({ name, visibility: "PRIVATE" });
      router.push(`/rooms/${room.id}?code=${encodeURIComponent(room.code)}`);
    } catch (err) {
      setError(err instanceof ApiError ? err.message : "Could not create room");
    } finally {
      setBusy(false);
    }
  }

  async function joinRoom(e: FormEvent) {
    e.preventDefault();
    setBusy(true);
    setError(null);
    try {
      const result = await roomsApi.joinByCode(code.trim().toUpperCase());
      const room = result.room;
      router.push(
        `/rooms/${room.id}?code=${encodeURIComponent(room.code)}`,
      );
    } catch (err) {
      setError(err instanceof ApiError ? err.message : "Could not join room");
    } finally {
      setBusy(false);
    }
  }

  if (loading || !user) {
    return (
      <Shell>
        <p className="text-slate-400">Loadingâ€¦</p>
      </Shell>
    );
  }

  return (
    <Shell>
      <Header
        title="Rooms"
        right={
          <div className="flex items-center gap-3">
            <span className="hidden text-sm text-slate-400 sm:inline">
              {user.name}
            </span>
            <Button variant="ghost" onClick={() => void logout()}>
              Log out
            </Button>
          </div>
        }
      />

      {error && (
        <div className="mb-4">
          <Alert>{error}</Alert>
        </div>
      )}

      <div className="grid gap-6 md:grid-cols-2">
        <form className="card space-y-4 p-6" onSubmit={createRoom}>
          <h2 className="text-lg font-medium text-white">Create a room</h2>
          <p className="text-sm text-slate-400">
            You become the host. Share the room code so friends can join.
          </p>
          <Field label="Room name">
            <Input
              required
              maxLength={100}
              value={name}
              onChange={(e) => setName(e.target.value)}
            />
          </Field>
          <Button type="submit" disabled={busy} className="w-full">
            Create private room
          </Button>
        </form>

        <form className="card space-y-4 p-6" onSubmit={joinRoom}>
          <h2 className="text-lg font-medium text-white">Join with code</h2>
          <p className="text-sm text-slate-400">
            Enter the invite code from the host.
          </p>
          <Field label="Room code">
            <Input
              required
              minLength={4}
              maxLength={32}
              placeholder="e.g. ABCD2345"
              value={code}
              onChange={(e) => setCode(e.target.value.toUpperCase())}
            />
          </Field>
          <Button type="submit" variant="ghost" disabled={busy} className="w-full">
            Join room
          </Button>
        </form>
      </div>
    </Shell>
  );
}

"use client";

import { useEffect, useMemo, useRef } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { useAuth } from "@/context/AuthContext";
import { useSocketConnection } from "@/hooks/useSocketConnection";
import { useRoomSession } from "@/hooks/useRoomSession";
import { usePlaybackSync } from "@/hooks/usePlaybackSync";
import { useChat } from "@/hooks/useChat";
import { PlayerPanel } from "@/components/room/PlayerPanel";
import { ChatPanel } from "@/components/room/ChatPanel";
import { MembersPanel } from "@/components/room/MembersPanel";
import {
  Alert,
  Button,
  Header,
  Shell,
  StatusDot,
} from "@/components/ui";

export function RoomClient({ roomId }: { roomId: string }) {
  const search = useSearchParams();
  const code = search.get("code") ?? undefined;
  const router = useRouter();
  const { user, loading } = useAuth();
  const videoRef = useRef<HTMLVideoElement | null>(null);

  const enabled = Boolean(user && roomId);
  const { socket, status } = useSocketConnection(enabled);

  const session = useRoomSession({ socket, roomId, code });

  const playback = usePlaybackSync({
    socket,
    roomId,
    role: session.myRole,
    videoRef,
  });

  const chat = useChat({
    socket,
    roomId,
    enabled: session.joined,
  });

  useEffect(() => {
    if (!loading && !user) router.replace("/login");
  }, [loading, user, router]);

  useEffect(() => {
    if (session.kicked) router.replace("/rooms");
  }, [session.kicked, router]);

  const inviteHint = useMemo(() => session.room?.code ?? null, [session.room]);

  if (loading || !user) {
    return (
      <Shell>
        <p className="text-slate-400">Loadingâ€¦</p>
      </Shell>
    );
  }

  return (
    <Shell>
      <Header
        title={session.room?.name ?? "Room"}
        right={
          <div className="flex flex-wrap items-center gap-3">
            <StatusDot
              status={
                status === "connected"
                  ? "connected"
                  : status === "connecting"
                    ? "connecting"
                    : status === "error"
                      ? "error"
                      : "disconnected"
              }
            />
            <Button
              variant="ghost"
              onClick={() => {
                session.leavePermanently();
                router.push("/rooms");
              }}
            >
              Leave
            </Button>
          </div>
        }
      />

      {inviteHint && (
        <div className="mb-4 flex flex-wrap items-center gap-3 text-sm text-slate-300">
          <span>
            Invite code:{" "}
            <code className="rounded-lg bg-white/10 px-2 py-1 font-mono text-accent-soft">
              {inviteHint}
            </code>
          </span>
          <Button
            variant="ghost"
            className="text-xs"
            onClick={() => void navigator.clipboard.writeText(inviteHint)}
          >
            Copy code
          </Button>
          {session.myRole && (
            <span className="badge bg-white/10">You: {session.myRole}</span>
          )}
        </div>
      )}

      {session.error && (
        <div className="mb-4">
          <Alert>{session.error}</Alert>
        </div>
      )}

      {!session.joined && !session.error && (
        <p className="mb-4 text-sm text-slate-400">Joining roomâ€¦</p>
      )}

      <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
        <div className="space-y-4">
          <PlayerPanel
            videoRef={videoRef}
            state={playback.state}
            canControl={playback.canControl}
            role={session.myRole}
            needsGesture={playback.needsGesture}
            loadError={playback.loadError}
            error={playback.error}
            onCommand={playback.sendCommand}
            onResumeGesture={() => void playback.resumeFromGesture()}
          />
          <MembersPanel
            members={session.members}
            myRole={session.myRole}
            myUserId={user.id}
            hostId={session.room?.hostId}
            onTransfer={session.transferHost}
            onRemove={session.removeMember}
            onChangeRole={session.changeRole}
          />
        </div>
        <ChatPanel
          messages={chat.messages}
          typing={chat.typing}
          reactions={chat.reactions}
          error={chat.error}
          onSend={chat.send}
          onTyping={chat.setIsTyping}
          onReact={chat.react}
        />
      </div>
    </Shell>
  );
}

import type { ButtonHTMLAttributes, InputHTMLAttributes, ReactNode } from "react";

export function Button(
  props: ButtonHTMLAttributes<HTMLButtonElement> & {
    variant?: "primary" | "ghost" | "danger";
  },
) {
  const { variant = "primary", className = "", ...rest } = props;
  const v =
    variant === "ghost"
      ? "btn-ghost"
      : variant === "danger"
        ? "btn-danger"
        : "btn-primary";
  return <button className={`${v} ${className}`} {...rest} />;
}

export function Input(props: InputHTMLAttributes<HTMLInputElement>) {
  const { className = "", ...rest } = props;
  return <input className={`input ${className}`} {...rest} />;
}

export function Field({
  label,
  children,
}: {
  label: string;
  children: ReactNode;
}) {
  return (
    <label className="block space-y-1.5">
      <span className="text-xs font-medium text-slate-400">{label}</span>
      {children}
    </label>
  );
}

export function Alert({
  tone = "error",
  children,
}: {
  tone?: "error" | "info" | "ok";
  children: ReactNode;
}) {
  const colors =
    tone === "ok"
      ? "border-emerald-500/30 bg-emerald-500/10 text-emerald-200"
      : tone === "info"
        ? "border-sky-500/30 bg-sky-500/10 text-sky-200"
        : "border-rose-500/30 bg-rose-500/10 text-rose-200";
  return (
    <div className={`rounded-xl border px-3 py-2 text-sm ${colors}`}>
      {children}
    </div>
  );
}

export function StatusDot({
  status,
}: {
  status: "connected" | "connecting" | "disconnected" | "error" | "idle";
}) {
  const color =
    status === "connected"
      ? "bg-emerald-400"
      : status === "connecting"
        ? "bg-amber-400 animate-pulse"
        : status === "error"
          ? "bg-rose-400"
          : "bg-slate-500";
  return (
    <span className="inline-flex items-center gap-2 text-xs text-slate-400">
      <span className={`h-2 w-2 rounded-full ${color}`} />
      {status}
    </span>
  );
}

"use client";

import { FormEvent, useEffect, useRef, useState } from "react";
import type { ChatMessageView } from "@syncstream/contracts";
import { Button, Input } from "@/components/ui";

type Props = {
  messages: ChatMessageView[];
  typing: Array<{ userId: string; name: string }>;
  reactions: Array<{ id: string; name: string; emoji: string }>;
  error: string | null;
  onSend: (content: string) => void;
  onTyping: (isTyping: boolean) => void;
  onReact: (emoji: string) => void;
};

const EMOJIS = ["🔥", "😂", "👏", "❤️", "😮"];

export function ChatPanel({
  messages,
  typing,
  reactions,
  error,
  onSend,
  onTyping,
  onReact,
}: Props) {
  const [text, setText] = useState("");
  const bottomRef = useRef<HTMLDivElement>(null);
  const typingTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages.length]);

  function submit(e: FormEvent) {
    e.preventDefault();
    if (!text.trim()) return;
    onSend(text);
    setText("");
    onTyping(false);
  }

  function onChange(value: string) {
    setText(value);
    onTyping(true);
    if (typingTimer.current) clearTimeout(typingTimer.current);
    typingTimer.current = setTimeout(() => onTyping(false), 1200);
  }

  return (
    <div className="card flex h-full min-h-[320px] flex-col">
      <div className="flex items-center justify-between border-b border-white/10 px-4 py-3">
        <h3 className="text-sm font-medium text-white">Chat</h3>
        <div className="flex gap-1">
          {EMOJIS.map((e) => (
            <button
              key={e}
              type="button"
              className="rounded-lg px-1.5 py-0.5 text-sm hover:bg-white/10"
              onClick={() => onReact(e)}
            >

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from "react";
import { authApi } from "@/lib/api";
import {
  clearSession,
  getStoredUser,
  setSession,
  type StoredUser,
} from "@/lib/auth-storage";
import { disconnectSocket } from "@/lib/socket";

type AuthState = {
  user: StoredUser | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  register: (name: string, email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
  refreshMe: () => Promise<void>;
};

const AuthContext = createContext<AuthState | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<StoredUser | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const cached = getStoredUser();
    if (cached) setUser(cached);
    authApi
      .me()
      .then((u) => setUser(u))
      .catch(() => {
        clearSession();
        setUser(null);
      })
      .finally(() => setLoading(false));
  }, []);

  const login = useCallback(async (email: string, password: string) => {
    const result = await authApi.login({ email, password });
    setSession({
      accessToken: result.tokens.accessToken,
      refreshToken: result.tokens.refreshToken,
      user: result.user,
    });
    setUser(result.user);
  }, []);

  const register = useCallback(
    async (name: string, email: string, password: string) => {
      const result = await authApi.register({ name, email, password });
      setSession({
        accessToken: result.tokens.accessToken,
        refreshToken: result.tokens.refreshToken,
        user: result.user,
      });
      setUser(result.user);
    },
    [],
  );

  const logout = useCallback(async () => {
    try {
      await authApi.logout();
    } finally {
      disconnectSocket();
      clearSession();
      setUser(null);
    }
  }, []);

  const refreshMe = useCallback(async () => {
    const u = await authApi.me();
    setUser(u);
  }, []);

  const value = useMemo(
    () => ({ user, loading, login, register, logout, refreshMe }),
    [user, loading, login, register, logout, refreshMe],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used within AuthProvider");
  return ctx;
}

"use client";

import { useCallback, useEffect, useState } from "react";
import type { ChatMessageView } from "@syncstream/contracts";
import type { AppSocket } from "@/lib/socket";

export function useChat(opts: {
  socket: AppSocket | null;
  roomId: string | null;
  enabled: boolean;
}) {
  const { socket, roomId, enabled } = opts;
  const [messages, setMessages] = useState<ChatMessageView[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [typing, setTyping] = useState<
    Array<{ userId: string; name: string }>
  >([]);
  const [reactions, setReactions] = useState<
    Array<{
      id: string;
      userId: string;
      name: string;
      emoji: string;
      at: number;
    }>
  >([]);

  useEffect(() => {
    if (!socket || !roomId || !enabled) return;

    socket.emit("chat:history", { roomId, limit: 50 });

    const onMessage = (msg: ChatMessageView) => {
      if (msg.roomId !== roomId) return;
      setMessages((prev) => {
        if (prev.some((m) => m.id === msg.id)) return prev;
        return [...prev, msg];
      });
    };
    const onHistory = (payload: {
      roomId: string;
      messages: ChatMessageView[];
      nextCursor: string | null;
    }) => {
      if (payload.roomId !== roomId) return;
      setMessages([...payload.messages].reverse());
    };
    const onError = (payload: {
      code: string;
      message: string;
      roomId?: string;
    }) => {
      if (payload.roomId && payload.roomId !== roomId) return;
      setError(payload.message);
    };
    const onTyping = (payload: {
      roomId: string;
      userId: string;
      name: string;
      isTyping: boolean;
    }) => {
      if (payload.roomId !== roomId) return;
      setTyping((prev) => {
        const without = prev.filter((t) => t.userId !== payload.userId);
        return payload.isTyping
          ? [...without, { userId: payload.userId, name: payload.name }]
          : without;
      });
    };
    const onReaction = (payload: {
      roomId: string;
      userId: string;
      name: string;
      emoji: string;
      at: number;
    }) => {
      if (payload.roomId !== roomId) return;
      const id = `${payload.userId}-${payload.at}`;
      setReactions((prev) => [
        ...prev.slice(-20),
        {
          id,
          userId: payload.userId,
          name: payload.name,
          emoji: payload.emoji,
          at: payload.at,
        },
      ]);
      setTimeout(() => {
        setReactions((prev) => prev.filter((r) => r.id !== id));
      }, 2500);
    };

    socket.on("chat:message", onMessage);
    socket.on("chat:history", onHistory);
    socket.on("chat:error", onError);
    socket.on("typing:update", onTyping);
    socket.on("reaction:new", onReaction);

    return () => {
      socket.off("chat:message", onMessage);
      socket.off("chat:history", onHistory);
      socket.off("chat:error", onError);
      socket.off("typing:update", onTyping);
      socket.off("reaction:new", onReaction);
    };
  }, [socket, roomId, enabled]);

  const send = useCallback(
    (content: string) => {
      if (!socket || !roomId || !content.trim()) return;
      setError(null);
      socket.emit("chat:send", {
        roomId,
        content: content.trim(),
        clientMessageId: crypto.randomUUID(),
      });
    },
    [socket, roomId],
  );

  const setIsTyping = useCallback(
    (isTyping: boolean) => {
      if (!socket || !roomId) return;
      socket.emit("typing:update", { roomId, isTyping });
    },
    [socket, roomId],
  );

  const react = useCallback(
    (emoji: string) => {
      if (!socket || !roomId) return;
      socket.emit("reaction:send", { roomId, emoji });
    },
    [socket, roomId],
  );

  return { messages, error, typing, reactions, send, setIsTyping, react };
}

"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import type { PlaybackState, RoomRole } from "@syncstream/contracts";
import type { AppSocket } from "@/lib/socket";
import { decideDriftCorrection, expectedPosition } from "@/lib/playback/math";
import {
  HtmlVideoPlayer,
  type VideoPlayer,
} from "@/lib/playback/video-player";

const SAMPLE_MS = 400;
const RATE_RESET_MS = 2_000;

function emptyState(): PlaybackState {
  return {
    videoId: "",
    status: "PAUSED",
    position: 0,
    updatedAt: Date.now(),
    playbackRate: 1,
    sequence: 0,
  };
}

export function usePlaybackSync(opts: {
  socket: AppSocket | null;
  roomId: string | null;
  role: RoomRole | null;
  videoRef: React.RefObject<HTMLVideoElement | null>;
}) {
  const { socket, roomId, role, videoRef } = opts;
  const [state, setState] = useState<PlaybackState>(emptyState());
  const [needsGesture, setNeedsGesture] = useState(false);
  const [loadError, setLoadError] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const appliedSeq = useRef(0);
  const playerRef = useRef<VideoPlayer | null>(null);
  const bufferingRef = useRef(false);
  const rateResetTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const applyingRef = useRef(false);

  const canControl = role === "HOST" || role === "CO_HOST";

  useEffect(() => {
    const el = videoRef.current;
    if (!el) return;
    playerRef.current = new HtmlVideoPlayer(el);

    const onWaiting = () => {
      bufferingRef.current = true;
    };
    const onPlaying = () => {
      bufferingRef.current = false;
      setNeedsGesture(false);
    };
    const onError = () => setLoadError("Failed to load video");

    el.addEventListener("waiting", onWaiting);
    el.addEventListener("playing", onPlaying);
    el.addEventListener("error", onError);
    return () => {
      el.removeEventListener("waiting", onWaiting);
      el.removeEventListener("playing", onPlaying);
      el.removeEventListener("error", onError);
    };
  }, [videoRef]);

  const applyState = useCallback(async (next: PlaybackState) => {
    if (next.sequence < appliedSeq.current) return;
    appliedSeq.current = next.sequence;
    setState(next);

    const player = playerRef.current;
    if (!player) return;

    applyingRef.current = true;
    try {
      if (next.videoId && player.getSrc() !== next.videoId) {
        setLoadError(null);
        player.setSrc(next.videoId);
      }

      const now = Date.now();
      const target = expectedPosition(next, now);
      player.setPlaybackRate(next.playbackRate || 1);

      const local = player.getCurrentTime();
      if (Math.abs(local - target) > 0.35) {
        player.seek(target);
      }

      if (next.status === "PLAYING") {
        try {
          await player.play();
          setNeedsGesture(false);
        } catch {
          setNeedsGesture(true);
        }
      } else {
        player.pause();
      }
    } finally {
      setTimeout(() => {
        applyingRef.current = false;
      }, 250);
    }
  }, []);

  useEffect(() => {
    if (!socket || !roomId) return;

    const onState = (payload: { roomId: string; state: PlaybackState }) => {
      if (payload.roomId !== roomId) return;
      void applyState(payload.state);
    };
    const onSnapshot = (payload: { roomId: string; state: PlaybackState }) => {
      if (payload.roomId !== roomId) return;
      void applyState(payload.state);
    };
    const onError = (payload: {
      code: string;
      message: string;
      roomId?: string;
    }) => {
      if (payload.roomId && payload.roomId !== roomId) return;
      setError(payload.message);
    };

    socket.on("playback:state", onState);
    socket.on("playback:snapshot", onSnapshot);
    socket.on("playback:error", onError);
    socket.emit("playback:sync", { roomId });

    return () => {
      socket.off("playback:state", onState);
      socket.off("playback:snapshot", onSnapshot);
      socket.off("playback:error", onError);
    };
  }, [socket, roomId, applyState]);

  useEffect(() => {
    if (!roomId) return;
    const id = setInterval(() => {
      const player = playerRef.current;
      if (!player || !state.videoId) return;
      if (applyingRef.current || bufferingRef.current || needsGesture) return;
      if (state.status !== "PLAYING") {
        player.setPlaybackRate(state.playbackRate || 1);
        return;
      }

      const expected = expectedPosition(state, Date.now());
      const local = player.getCurrentTime();
      const action = decideDriftCorrection(
        local,
        expected,
        state.playbackRate || 1,
      );

      if (action.kind === "rate") {
        player.setPlaybackRate(action.rate);
        if (rateResetTimer.current) clearTimeout(rateResetTimer.current);
        rateResetTimer.current = setTimeout(() => {
          player.setPlaybackRate(state.playbackRate || 1);
        }, RATE_RESET_MS);
      } else if (action.kind === "seek") {
        player.seek(action.position);
        player.setPlaybackRate(state.playbackRate || 1);
      }
    }, SAMPLE_MS);

    return () => {
      clearInterval(id);
      if (rateResetTimer.current) clearTimeout(rateResetTimer.current);
    };
  }, [roomId, state, needsGesture]);

  const sendCommand = useCallback(
    (cmd: {
      action: "PLAY" | "PAUSE" | "SEEK" | "CHANGE_VIDEO";
      position?: number;
      videoId?: string;
    }) => {
      if (!socket || !roomId || !canControl) return;
      setError(null);
      socket.emit("playback:command", {
        roomId,
        action: cmd.action,
        position: cmd.position,
        videoId: cmd.videoId,
        commandId: crypto.randomUUID(),
      });
    },
    [socket, roomId, canControl],
  );

  const resumeFromGesture = useCallback(async () => {
    const player = playerRef.current;
    if (!player) return;
    try {
      await player.play();
      setNeedsGesture(false);
    } catch {
      setNeedsGesture(true);
    }
  }, []);

  const requestSync = useCallback(() => {
    if (!socket || !roomId) return;
    socket.emit("playback:sync", { roomId });
  }, [socket, roomId]);

  return {
    state,
    canControl,
    needsGesture,
    loadError,
    error,
    sendCommand,
    resumeFromGesture,
    requestSync,
  };
}

  const myRole: RoomRole | null =
    room?.myRole ??
    members.find((m) => m.userId === user?.id)?.role ??
    null;

  useEffect(() => {
    if (!socket || !roomId) return;

    setError(null);
    setKicked(false);
    socket.emit("room:join", {
      roomId,
      ...(code ? { code } : {}),
    });

    const onState = (payload: {
      room: RoomView;
      members: RoomMemberView[];
    }) => {
      setRoom(payload.room);
      setMembers(payload.members);
      setJoined(true);
    };
    const onMembers = (payload: {
      roomId: string;
      members: RoomMemberView[];
    }) => {
      if (payload.roomId !== roomId) return;
      setMembers(payload.members);
    };
    const onError = (payload: {
      code: string;
      message: string;
      roomId?: string;
    }) => {
      if (payload.roomId && payload.roomId !== roomId) return;
      setError(payload.message);
    };
    const onHostChanged = (payload: {
      roomId: string;
      previousHostId: string;
      newHostId: string;
      reason: string;
    }) => {
      if (payload.roomId !== roomId) return;
      setRoom((prev) =>
        prev
          ? {
              ...prev,
              hostId: payload.newHostId,
              myRole:
                user?.id === payload.newHostId
                  ? "HOST"
                  : prev.myRole === "HOST"
                    ? "VIEWER"
                    : prev.myRole,
            }
          : prev,
      );
    };
    const onRemoved = (payload: {
      roomId: string;
      userId: string;
      removedBy: string;
    }) => {
      if (payload.roomId !== roomId) return;
      if (payload.userId === user?.id) {
        setKicked(true);
        setJoined(false);
      }
    };
    const onRoleChanged = (payload: {
      roomId: string;
      userId: string;
      role: RoomRole;
    }) => {
      if (payload.roomId !== roomId) return;
      if (payload.userId === user?.id) {
        setRoom((prev) =>
          prev ? { ...prev, myRole: payload.role } : prev,
        );
      }
    };

    socket.on("room:state", onState);
    socket.on("room:members", onMembers);
    socket.on("room:error", onError);
    socket.on("room:host-changed", onHostChanged);
    socket.on("room:member-removed", onRemoved);
    socket.on("room:member-role-changed", onRoleChanged);

    const onConnect = () => {
      socket.emit("room:join", {
        roomId,
        ...(code ? { code } : {}),
      });
    };
    socket.on("connect", onConnect);

    return () => {
      socket.off("room:state", onState);
      socket.off("room:members", onMembers);
      socket.off("room:error", onError);
      socket.off("room:host-changed", onHostChanged);
      socket.off("room:member-removed", onRemoved);
      socket.off("room:member-role-changed", onRoleChanged);
      socket.off("connect", onConnect);
      socket.emit("room:leave", { roomId, leaveMembership: false });
    };
  }, [socket, roomId, code, user?.id]);

  const leavePermanently = useCallback(() => {
    if (!socket) return;
    socket.emit("room:leave", { roomId, leaveMembership: true });
    setJoined(false);
  }, [socket, roomId]);

  const transferHost = useCallback(
    (toUserId: string) => {
      if (!socket) return;
      socket.emit("room:transfer-host", { roomId, toUserId });
    },
    [socket, roomId],
  );

  const removeMember = useCallback(
    (userId: string) => {
      if (!socket) return;
      socket.emit("room:remove-member", { roomId, userId });
    },
    [socket, roomId],
  );

  const changeRole = useCallback(
    (userId: string, role: "CO_HOST" | "VIEWER") => {
      if (!socket) return;
      socket.emit("room:change-role", { roomId, userId, role });
    },
    [socket, roomId],
  );

  return {
    room,
    members,
    joined,
    myRole,
    error,
    kicked,
    leavePermanently,
    transferHost,
    removeMember,
    changeRole,
  };
}

"use client";

import { useEffect, useRef, useState } from "react";
import { connectSocket, disconnectSocket, type AppSocket } from "@/lib/socket";
import { getAccessToken } from "@/lib/auth-storage";

export type ConnectionStatus =
  | "idle"
  | "connecting"
  | "connected"
  | "disconnected"
  | "error";

export function useSocketConnection(enabled: boolean) {
  const [status, setStatus] = useState<ConnectionStatus>("idle");
  const [socket, setSocket] = useState<AppSocket | null>(null);
  const [error, setError] = useState<string | null>(null);
  const enabledRef = useRef(enabled);
  enabledRef.current = enabled;

  useEffect(() => {
    if (!enabled) {
      setStatus("idle");
      setSocket(null);
      return;
    }
    if (!getAccessToken()) {
      setStatus("error");
      setError("Not authenticated");
      return;
    }

    let s: AppSocket;
    try {
      setStatus("connecting");
      s = connectSocket();
    } catch (e) {
      setStatus("error");
      setError(e instanceof Error ? e.message : "Socket connect failed");
      return;
    }

    const onConnect = () => {
      setStatus("connected");
      setError(null);
      setSocket(s);
    };
    const onDisconnect = () => setStatus("disconnected");
    const onConnectError = (err: Error) => {
      setStatus("error");
      setError(err.message);
    };

    s.on("connect", onConnect);
    s.on("disconnect", onDisconnect);
    s.on("connect_error", onConnectError);
    if (s.connected) onConnect();
    setSocket(s);

    return () => {
      s.off("connect", onConnect);
      s.off("disconnect", onDisconnect);
      s.off("connect_error", onConnectError);
      if (!enabledRef.current) disconnectSocket();
    };
  }, [enabled]);

  return { socket, status, error };
}

export class ApiError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
  ) {
    super(message);
    this.name = "ApiError";
  }
}

type ApiSuccess<T> = {
  success: true;
  message?: string;
  data: T;
};

type ApiFail = {
  success: false;
  error: { code: string; message: string; details?: unknown };
};

async function parseJson(res: Response) {
  try {
    return await res.json();
  } catch {
    return null;
  }
}

let refreshInFlight: Promise<string | null> | null = null;

async function refreshAccessToken(): Promise<string | null> {
  if (refreshInFlight) return refreshInFlight;
  refreshInFlight = (async () => {
    const refreshToken = getRefreshToken();
    if (!refreshToken) return null;
    try {
      const res = await fetch(`${config.apiUrl}/auth/refresh`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ refreshToken }),
      });
      const body = (await parseJson(res)) as
        | ApiSuccess<{
            accessToken: string;
            refreshToken: string;
            expiresIn: string;
          }>
        | ApiFail
        | null;
      if (!res.ok || !body || !("success" in body) || !body.success) {
        clearSession();
        return null;
      }
      setSession({
        accessToken: body.data.accessToken,
        refreshToken: body.data.refreshToken,
      });
      return body.data.accessToken;
    } catch {
      return null;
    } finally {
      refreshInFlight = null;
    }
  })();
  return refreshInFlight;
}

export async function apiFetch<T>(
  path: string,
  init: RequestInit = {},
  opts: { auth?: boolean; retry?: boolean } = {},
): Promise<T> {
  const { auth = true, retry = true } = opts;
  const headers = new Headers(init.headers);
  if (!headers.has("Content-Type") && init.body) {
    headers.set("Content-Type", "application/json");
  }
  if (auth) {
    const token = getAccessToken();
    if (token) headers.set("Authorization", `Bearer ${token}`);
  }

  const res = await fetch(`${config.apiUrl}${path}`, { ...init, headers });
  const body = await parseJson(res);

  if (res.status === 401 && auth && retry) {
    const next = await refreshAccessToken();
    if (next) {
      setAccessToken(next);
      return apiFetch<T>(path, init, { auth, retry: false });
    }
  }

  if (!res.ok) {
    const fail = body as ApiFail | null;
    throw new ApiError(
      res.status,
      fail?.error?.code ?? "HTTP_ERROR",
      (fail?.error?.message ?? res.statusText) || "Request failed",
    );
  }

  if (body && typeof body === "object" && "success" in body && body.success) {
    return (body as ApiSuccess<T>).data;
  }
  return body as T;
}

export type StoredUser = {
  id: string;
  email: string;
  name: string;
  avatarUrl: string | null;
};

export function getAccessToken(): string | null {
  if (typeof window === "undefined") return null;
  return localStorage.getItem(ACCESS);
}

export function getRefreshToken(): string | null {
  if (typeof window === "undefined") return null;
  return localStorage.getItem(REFRESH);
}

export function getStoredUser(): StoredUser | null {
  if (typeof window === "undefined") return null;
  const raw = localStorage.getItem(USER);
  if (!raw) return null;
  try {
    return JSON.parse(raw) as StoredUser;
  } catch {
    return null;
  }
}

export function setSession(opts: {
  accessToken: string;
  refreshToken: string;
  user?: StoredUser;
}) {
  localStorage.setItem(ACCESS, opts.accessToken);
  localStorage.setItem(REFRESH, opts.refreshToken);
  if (opts.user) localStorage.setItem(USER, JSON.stringify(opts.user));
}

export function setAccessToken(token: string) {
  localStorage.setItem(ACCESS, token);
}

export function clearSession() {
  localStorage.removeItem(ACCESS);
  localStorage.removeItem(REFRESH);
  localStorage.removeItem(USER);
}

import type { PlaybackState } from "@syncstream/contracts";

export function expectedPosition(state: PlaybackState, nowMs: number): number {
  if (state.status === "PAUSED") return Math.max(0, state.position);
  const elapsedSec = (nowMs - state.updatedAt) / 1000;
  return Math.max(0, state.position + state.playbackRate * elapsedSec);
}

export type DriftAction =
  | { kind: "none" }
  | { kind: "rate"; rate: number }
  | { kind: "seek"; position: number };

export function decideDriftCorrection(
  localPosition: number,
  expected: number,
  baseRate: number,
): DriftAction {
  const driftSec = localPosition - expected;
  const abs = Math.abs(driftSec);

  if (abs < 0.15) return { kind: "none" };

  if (abs <= 0.5) {
    const adj = driftSec < 0 ? 1.04 : 0.96;
    return { kind: "rate", rate: baseRate * adj };
  }

  return { kind: "seek", position: expected };
}

export interface VideoPlayer {
  getCurrentTime(): number;
  isPlaying(): boolean;
  play(): Promise<void>;
  pause(): void;
  seek(position: number): void;
  setPlaybackRate(rate: number): void;
  setSrc(url: string): void;
  getSrc(): string;
  ready(): boolean;
  on(event: string, handler: () => void): void;
  off(event: string, handler: () => void): void;
}

export class HtmlVideoPlayer implements VideoPlayer {
  constructor(private readonly el: HTMLVideoElement) {}

  getCurrentTime(): number {
    return this.el.currentTime || 0;
  }

  isPlaying(): boolean {
    return !this.el.paused && !this.el.ended;
  }

  async play(): Promise<void> {
    try {
      await this.el.play();
    } catch {
      throw new Error("PLAY_BLOCKED");
    }
  }

  pause(): void {
    this.el.pause();
  }

  seek(position: number): void {
    if (!Number.isFinite(position) || position < 0) return;
    const wasPlaying = this.isPlaying();
    this.el.currentTime = position;
    if (wasPlaying) void this.play().catch(() => undefined);
  }

  setPlaybackRate(rate: number): void {
    if (!Number.isFinite(rate) || rate <= 0) return;
    this.el.playbackRate = rate;
  }

  setSrc(url: string): void {
    if (this.el.src === url) return;
    this.el.src = url;
    this.el.load();
  }

  getSrc(): string {
    return this.el.currentSrc || this.el.src || "";
  }

  ready(): boolean {
    return this.el.readyState >= HTMLMediaElement.HAVE_CURRENT_DATA;
  }

  on(event: string, handler: () => void): void {
    this.el.addEventListener(event, handler);
  }

  off(event: string, handler: () => void): void {
    this.el.removeEventListener(event, handler);
  }
}

 * The server validates, checks permissions, deduplicates by commandId,
 * then produces a new PlaybackState.
 */
export const PlaybackActionSchema = z.enum([
  "PLAY",
  "PAUSE",
  "SEEK",
  "CHANGE_VIDEO",
]);
export type PlaybackAction = z.infer<typeof PlaybackActionSchema>;

export const PlaybackCommandSchema = z
  .object({
    roomId: z.uuid(),
    action: PlaybackActionSchema,
    position: z.number().min(0).optional(), // required for SEEK, optional for PLAY
    videoId: z.string().min(1).optional(), // required for CHANGE_VIDEO
    commandId: z.uuid(), // client-generated, used for dedup
  })
  .superRefine((val, ctx) => {
    if (val.action === "SEEK" && val.position === undefined) {
      ctx.addIssue({
        code: "custom",
        message: "position is required for SEEK",
        path: ["position"],
      });
    }
    if (val.action === "CHANGE_VIDEO" && !val.videoId) {
      ctx.addIssue({
        code: "custom",
        message: "videoId is required for CHANGE_VIDEO",
        path: ["videoId"],
      });
    }
  });
export type PlaybackCommand = z.infer<typeof PlaybackCommandSchema>;

/** Client asks for the current authoritative state (reconnect / late join). */

import type { Request, Response, NextFunction, RequestHandler } from "express";

export function asyncHandler(
  fn: (req: Request, res: Response, next: NextFunction) => Promise<unknown>,
): RequestHandler {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

export class AppError extends Error {
  constructor(
    public readonly code: string,
    message: string,
    public readonly statusCode: number = 500,
    public readonly details?: unknown,
  ) {
    super(message);
    this.name = "AppError";
    Object.setPrototypeOf(this, new.target.prototype);
  }
}

export class NotFoundError extends AppError {
  constructor(resource: string, id?: string) {
    super(
      "NOT_FOUND",
      id ? `${resource} with id ${id} not found` : `${resource} not found`,
      404,
    );
  }
}

export class BadRequestError extends AppError {
  constructor(message = "Bad Request", details?: unknown) {
    super("BAD_REQUEST", message, 400, details);
  }
}

export class ValidationError extends AppError {
  constructor(message: string, details?: unknown) {
    super("VALIDATION_ERROR", message, 400, details);
  }
}

export class UnauthorizedError extends AppError {
  constructor(message = "Unauthorized", details?: unknown) {
    super("UNAUTHORIZED", message, 401, details);
  }
}

export class ConflictError extends AppError {
  constructor(message: string, details?: unknown) {
    super("CONFLICT", message, 409, details);
  }
}

  constructor(message = "Forbidden", details?: unknown) {
    super("FORBIDDEN", message, 403, details);
  }
}
